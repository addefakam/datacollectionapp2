   164→   164→function formatValue(unit: string, v: number): string {
   165→   165→  if (unit === 'ETB') return `ETB ${Math.round(v).toLocaleString()}`;
   166→   166→  const n = Math.round(v);
   167→   167→  const word =
   168→   168→    unit === 'enterprises' ? (n === 1 ? 'enterprise' : 'enterprises')
   169→   169→    : unit === 'jobs' ? (n === 1 ? 'job' : 'jobs')
   170→   170→    : unit;
   171→   171→  return `${n.toLocaleString()} ${word}`;
   172→   172→}
   173→   173→
   174→   174→/** Build the refDate payload for the API from the form's period-type inputs. */
   175→   175→function refDateFrom(periodType: string, value: string): string {
   176→   176→  if (periodType === 'weekly') return `${value}T12:00:00`;
   177→   177→  if (periodType === 'monthly') {
   178→   178→    const [y, m] = value.split('-').map(Number);
   179→   179→    return new Date(y, m - 1, 15).toISOString();
   180→   180→  }
   181→   181→  return new Date(Number(value), 6, 1).toISOString(); // mid-year for yearly
   182→   182→}
   183→   183→
   184→   184→function todayString(): string {
   185→   185→  const d = new Date();
   186→   186→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
   187→   187→}
   188→   188→
   189→   189→function currentMonthString(): string {
   190→   190→  return todayString().slice(0, 7);
   191→   191→}
   192→   192→
   193→   193→/** Derive the form's reference value from a stored periodStart (for editing). */
   194→   194→function refValueFrom(goal: Goal): string {
   195→   195→  const d = new Date(goal.periodStart);
   196→   196→  if (goal.periodType === 'weekly') {
   197→   197→    return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
   198→   198→  }
   199→   199→  if (goal.periodType === 'monthly') {
   200→   200→    return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
   201→   201→  }
   202→   202→  return String(d.getFullYear());
   203→   203→}

   688→   688→      map.set(g.periodType, list);
   689→   689→    }
   690→   690→    for (const list of map.values()) {
   691→   691→      list.sort((a, b) => (a.progress.status === 'achieved' ? 1 : 0) - (b.progress.status === 'achieved' ? 1 : 0) || (a.periodStart < b.periodStart ? 1 : -1));
   692→   692→    }
   693→   693→    return map;
   694→   694→  }, [goals]);

   701→   701→          <div className="pointer-events-none absolute -right-10 -top-16 h-56 w-56 rounded-full bg-white/10 blur-2xl" />
   702→   702→          <div className="pointer-events-none absolute -bottom-20 right-24 h-48 w-48 rounded-full bg-amber-300/20 blur-2xl" />
   703→   703→          <div className="relative flex flex-wrap items-center justify-between gap-3">
   704→   704→            <div className="min-w-0">
   705→   705→              <p className="flex items-center gap-1.5 text-[11px] font-bold uppercase tracking-wider text-emerald-50/80">

   707→   707→                {metricDefs.heroLabel}

   733→   733→                }}
   734→   734→              >
   735→   735→                <Plus className="h-4 w-4" />
   736→   736→                New Goal
