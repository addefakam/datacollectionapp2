     1→'use client';
     2→
     3→import { useState } from 'react';
     4→import { Button } from '@/components/ui/button';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import {
     8→  Select,
     9→  SelectContent,
    10→  SelectItem,
    11→  SelectTrigger,
    12→  SelectValue,
    13→} from '@/components/ui/select';
    14→import {
    15→  Card,
    16→  CardContent,
    17→  CardDescription,
    18→  CardHeader,
    19→  CardTitle,
    20→} from '@/components/ui/card';
    21→import { useToast } from '@/hooks/use-toast';
    22→import {
    23→  Store,
    24→  Loader2,
    25→  RotateCcw,
    26→  ChevronLeft,
    27→  ChevronRight,
    28→  Check,
    29→  UserCircle,
    30→  Phone,
    31→  FileText,
    32→  Eye,
    33→} from 'lucide-react';
    34→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group';
    35→import { SMALL_ENTERPRISE_TYPES, SMALL_ENTERPRISE_OWNERSHIP_TYPES } from '@/lib/location-data';
    36→import { useSettings } from '@/components/SettingsProvider';
    37→
    38→interface SmallEnterpriseSurveyFormProps {
    39→  onSubmit: () => void;
    40→  surveyorName: string;
    41→  surveyorId: string;
    42→  isOnline?: boolean;
    43→  onOfflineSave?: (data: Record<string, unknown>) => Promise<boolean>;
    44→}
    45→
    46→const STEPS = [
    47→  { id: 1, title: 'Enterprise', titleOr: 'Midhaalee', icon: Store },
    48→  { id: 2, title: 'Capacity', titleOr: 'Dandeettii', icon: FileText },
    49→  { id: 3, title: 'Contact', titleOr: 'Quunnamtii', icon: Phone },
    50→  { id: 4, title: 'Review', titleOr: 'Mirkanaa', icon: Eye },
    51→];
    52→
    53→export default function SmallEnterpriseSurveyForm({
    54→  onSubmit,
    55→  surveyorName,
    56→  surveyorId,
    57→  isOnline = true,
    58→  onOfflineSave,
    59→}: SmallEnterpriseSurveyFormProps) {
    60→  const { toast } = useToast();
    61→  const { subCities, areasFor } = useSettings();
    62→  const [isSubmitting, setIsSubmitting] = useState(false);
    63→  const [currentStep, setCurrentStep] = useState(1);
    64→  const [selectedSubCity, setSelectedSubCity] = useState('');
    65→  const [areas, setAreas] = useState<string[]>([]);
    66→  const [dataConfirmed, setDataConfirmed] = useState('');
    67→
    68→  const emptyForm = {
    69→    enterpriseName: '',
    70→    enterpriseType: '',
    71→    ownershipType: '',
    72→    subCity: '',
    73→    area: '',
    74→    specificAddress: '',
    75→    licenseNumber: '',
    76→    employeeCount: '',
    77→    maleEmployees: '',
    78→    femaleEmployees: '',
    79→    startupCapital: '',
    80→    hasPremises: 'no',
    81→    hasTraining: 'no',
    82→    contactName: '',
    83→    contactPhone: '',
    84→  };
    85→
    86→  const [formData, setFormData] = useState(emptyForm);
    87→
    88→  const handleSubCityChange = (value: string) => {
    89→    setSelectedSubCity(value);
    90→    setAreas(areasFor(value));
    91→    setFormData((prev) => ({ ...prev, subCity: value, area: '' }));
    92→  };
    93→
    94→  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
    95→    const { name, value } = e.target;
    96→    setFormData((prev) => ({ ...prev, [name]: value }));
    97→  };
    98→
    99→  const handleSelectChange = (name: string, value: string) => {
   100→    setFormData((prev) => ({ ...prev, [name]: value }));
   101→  };
   102→
   103→  /**
   104→   * Sex breakdown inputs: when both male and female are filled and the total
   105→   * is still empty, auto-fill the total with their sum.
   106→   */
   107→  const handleSexChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   108→    const { name, value } = e.target;
   109→    setFormData((prev) => {
   110→      const next = { ...prev, [name]: value } as typeof prev;
   111→      const male = name === 'maleEmployees' ? value : next.maleEmployees;
   112→      const female = name === 'femaleEmployees' ? value : next.femaleEmployees;
   113→      if (next.employeeCount === '' && male !== '' && female !== '') {
   114→        const sum = Number(male) + Number(female);
   115→        if (Number.isFinite(sum) && sum >= 0) return { ...next, employeeCount: String(sum) };
   116→      }
   117→      return next;
   118→    });
   119→  };
   120→
   121→  /** Shared step validation for one male/female pair against its total. */
   122→  const validateSexPair = (maleStr: string, femaleStr: string, totalStr: string, totalLabel: string): boolean => {
   123→    const male = maleStr === '' ? 0 : Number(maleStr);
   124→    const female = femaleStr === '' ? 0 : Number(femaleStr);
   125→    if (!Number.isFinite(male) || male < 0 || !Number.isFinite(female) || female < 0) {
   126→      toast({ title: 'Invalid Sex Counts / Lakkoofsa Sanyii Dogoggora', description: `Male and female ${totalLabel.toLowerCase()} must be non-negative numbers / Lakkoofsi dhiira fi dubartii lakkoofsa diidaa hin qabaatu`, variant: 'destructive' });
   127→      return false;
   128→    }
   129→    if (male + female > 0 && male + female !== Number(totalStr)) {
   130→      toast({ title: 'Sex Counts Mismatch / Lakkoofsa Sanyii Hin Wal Qabne', description: `Male + Female (${male + female}) must equal ${totalLabel} (${totalStr}) / Dhiira fi Dubartii lakkoofsa waliigalaa wajjin wal qabachuu qabu`, variant: 'destructive' });
   131→      return false;
   132→    }
   133→    return true;
   134→  };
   135→
   136→  const validateStep = (step: number): boolean => {
   137→    switch (step) {
   138→      case 1:
   139→        if (!formData.enterpriseName || !formData.enterpriseType || !formData.subCity || !formData.area || !formData.specificAddress) {
   140→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: 'All fields in this step are required / Galii hundinuu barbaachisaadha', variant: 'destructive' });
   141→          return false;
   142→        }
   143→        return true;
   144→      case 2:
   145→        if (!formData.ownershipType) {
   146→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: 'Ownership type is required / Gosa abbaa qabuu barbaachisaadha', variant: 'destructive' });
   147→          return false;
   148→        }
   149→        if (!formData.employeeCount || Number(formData.employeeCount) < 0) {
   150→          toast({ title: 'Invalid Employee Count / Lakkoofsa Hojjetaa Dogoggora', description: 'Enter a valid employee count / Lakkoofsa hojjetaa sirrii galchi', variant: 'destructive' });
   151→          return false;
   152→        }
   153→        if (!validateSexPair(formData.maleEmployees, formData.femaleEmployees, formData.employeeCount, 'Employee Count')) return false;
   154→        if (formData.startupCapital !== '' && (Number.isNaN(Number(formData.startupCapital)) || Number(formData.startupCapital) < 0)) {
   155→          toast({ title: 'Invalid Capital / Maallaqa Dogoggora', description: 'Startup capital cannot be negative / Maallaqa jalqabaa diidaa hin qabaatu', variant: 'destructive' });
   156→          return false;
   157→        }
   158→        return true;
   159→      case 3:
   160→        if (!formData.contactName || !formData.contactPhone) {
   161→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: 'Contact person and phone are required / Maqaa fi bilbila quunnamtii barbaachisa', variant: 'destructive' });
   162→          return false;
   163→        }
   164→        {
   165→          const phoneClean = formData.contactPhone.replace(/[\s-]/g, '');
   166→          const ethPhoneRegex = /^(\+251|251|0)?(9|7)\d{8}$/;
   167→          if (!ethPhoneRegex.test(phoneClean)) {
   168→            toast({ title: 'Invalid Phone / Bilbila Dogoggora', description: 'Enter valid Ethiopian number (e.g., +251 91 234 5678)', variant: 'destructive' });
   169→            return false;
   170→          }
   171→        }
   172→        return true;
   173→      case 4:
   174→        if (dataConfirmed !== 'yes') {
   175→          toast({ title: 'Confirmation Required / Mirkanaa\u2019uu Barbaachisaa', description: 'Please confirm data is true and authenticated / Daataan dhugaa fi mirkanaa\u2019eef mirkanaadhu', variant: 'destructive' });
   176→          return false;
   177→        }
   178→        return true;
   179→      default:
   180→        return true;
   181→    }
   182→  };
   183→
   184→  const nextStep = () => { if (validateStep(currentStep)) setCurrentStep((s) => Math.min(s + 1, STEPS.length)); };
   185→  const prevStep = () => { setCurrentStep((s) => Math.max(s - 1, 1)); };
   186→
   187→  const resetForm = () => {
   188→    setFormData(emptyForm);
   189→    setSelectedSubCity('');
   190→    setAreas([]);
   191→    setDataConfirmed('');
   192→    setCurrentStep(1);
   193→  };
   194→
   195→  const handleSubmit = async (e: React.FormEvent) => {
   196→    e.preventDefault();
   197→    if (!validateStep(currentStep)) return;
   198→    const payload = {
   199→      ...formData,
   200→      employeeCount: Number(formData.employeeCount || 0),
   201→      maleEmployees: formData.maleEmployees === '' ? 0 : Number(formData.maleEmployees),
   202→      femaleEmployees: formData.femaleEmployees === '' ? 0 : Number(formData.femaleEmployees),
   203→      startupCapital: formData.startupCapital === '' ? null : Number(formData.startupCapital),
   204→      hasPremises: formData.hasPremises === 'yes',
   205→      hasTraining: formData.hasTraining === 'yes',
   206→      surveyorName,
   207→      surveyorId,
   208→    };
   209→    setIsSubmitting(true);
   210→    if (isOnline) {
   211→      try {
   212→        const response = await fetch('/api/smallenterprises', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
   213→        if (!response.ok) {
   214→          const err = await response.json().catch(() => ({}));
   215→          if (response.status >= 400 && response.status < 500) {
   216→            // Server permanently rejected the record (validation) — queueing it offline would never succeed.
   217→            toast({ title: 'Submission Rejected / Galmeen Morkame', description: err.error || 'Please correct the highlighted fields and submit again', variant: 'destructive' });
   218→            return;
   219→          }
   220→          throw new Error(err.error || 'Server unavailable');
   221→        }
   222→        toast({ title: 'Survey Submitted! / Galma Qabame!', description: `${formData.enterpriseName} recorded successfully / galmeen safiisan argame` });
   223→        resetForm();
   224→        onSubmit();
   225→      } catch (error) {
   226→        if (onOfflineSave) {
   227→          const saved = await onOfflineSave(payload);
   228→          if (saved) { toast({ title: 'Saved Offline / Offliin Kuusame', description: 'Will sync when connected / Netiirkii qunnamaa waanin dhiheenya' }); resetForm(); onSubmit(); }
   229→          else { toast({ title: 'Submission Failed / Galmeen Hin Dhufne', description: error instanceof Error ? error.message : 'Please try again / Dabalataan yaali', variant: 'destructive' }); }
   230→        } else { toast({ title: 'Submission Failed / Galmeen Hin Dhufne', description: error instanceof Error ? error.message : 'Please try again / Dabalataan yaali', variant: 'destructive' }); }
   231→      } finally { setIsSubmitting(false); }
   232→    } else {
   233→      if (onOfflineSave) {
   234→        const saved = await onOfflineSave(payload);
   235→        if (saved) { toast({ title: 'Saved Offline / Offliin Kuusame', description: 'Will sync when connected / Netiirkii qunnamaa waanin dhiheenya' }); resetForm(); onSubmit(); }
   236→        else { toast({ title: 'Save Failed / Hin Kuusanne', description: 'Storage full or unavailable / Qabiyyee dhiphaachaa ykn hin jiru', variant: 'destructive' }); }
   237→      }
   238→      setIsSubmitting(false);
   239→    }
   240→  };
   241→
   242→  const reviewItems = [
   243→    { label: 'Enterprise / Midhaalee', value: formData.enterpriseName },
   244→    { label: 'Enterprise Type / Gosa', value: formData.enterpriseType },
   245→    { label: 'Ownership / Abbaa Qabuu', value: formData.ownershipType },
   246→    { label: 'Sub-City / Kututtaa Maggalaa', value: formData.subCity },
   247→    { label: 'Werreda', value: formData.area },
   248→    { label: 'Address / Baka Addaa', value: formData.specificAddress },
   249→    { label: 'License No. / Lakofsaa Eyyeema', value: formData.licenseNumber || '—' },
   250→    { label: 'Employees / Hojjetaa', value: formData.employeeCount },
   251→    { label: 'Male / Dhiira', value: formData.maleEmployees === '' ? '—' : formData.maleEmployees },
   252→    { label: 'Female / Dubartii', value: formData.femaleEmployees === '' ? '—' : formData.femaleEmployees },
   253→    { label: 'Startup Capital (ETB)', value: formData.startupCapital === '' ? '—' : formData.startupCapital },
   254→    { label: 'Own Premises', value: formData.hasPremises === 'yes' ? 'Yes / Eyee' : 'No / Lakki' },
   255→    { label: 'MSE Training', value: formData.hasTraining === 'yes' ? 'Yes / Eyee' : 'No / Lakki' },
   256→    { label: 'Contact / Quunnamtii', value: `${formData.contactName} — ${formData.contactPhone}` },
   257→  ];
   258→
   259→  return (
   260→    <div className="mx-auto max-w-2xl">
   261→      {/* Step Indicator */}
   262→      <div className="mb-4 sm:mb-6">
   263→        <div className="flex items-center gap-1 overflow-x-auto pb-1 sm:gap-0 sm:pb-2" style={{ scrollbarWidth: 'none', msOverflowStyle: 'none' }}>
   264→          {STEPS.map((step, idx) => {
   265→            const Icon = step.icon;
   266→            const isActive = currentStep === step.id;
   267→            const isCompleted = currentStep > step.id;
   268→            return (
   269→              <div key={step.id} className="flex flex-1 shrink-0 items-center">
   270→                <button
   271→                  type="button"
   272→                  onClick={() => isCompleted && setCurrentStep(step.id)}
   273→                  className={`flex w-full flex-col items-center gap-0.5 rounded-lg px-1 py-1.5 transition-all sm:gap-1 sm:rounded-xl sm:px-2 sm:py-2 ${
   274→                    isActive ? 'bg-amber-600 text-white shadow-md sm:shadow-lg sm:shadow-amber-100'
   275→                      : isCompleted ? 'bg-amber-100 text-amber-700 cursor-pointer'
   276→                      : 'bg-gray-100 text-gray-400'
   277→                  }`}>
   278→                  <div className={`flex h-7 w-7 items-center justify-center rounded-full sm:h-8 sm:w-8 ${isCompleted ? 'bg-amber-600 text-white' : ''}`}>
   279→                    {isCompleted ? <Check className="h-3.5 w-3.5 sm:h-4 sm:w-4" /> : <Icon className="h-3.5 w-3.5 sm:h-4 sm:w-4" />}
   280→                  </div>
   281→                  <span className="text-[9px] leading-tight font-medium text-center truncate w-full sm:text-[10px]">{step.title}</span>
   282→                  <span className="hidden text-[8px] leading-tight text-center opacity-75 truncate w-full sm:block sm:text-[9px]">{step.titleOr}</span>
   283→                </button>
   284→                {idx < STEPS.length - 1 && (
   285→                  <div className={`mx-0.5 h-0.5 flex-1 rounded sm:mx-1 ${currentStep > step.id ? 'bg-amber-500' : 'bg-gray-200'}`} />
   286→                )}
   287→              </div>
   288→            );
   289→          })}
   290→        </div>
   291→        <div className="mt-1.5 flex items-center justify-between text-[11px] text-muted-foreground sm:mt-2 sm:text-xs">
   292→          <span>Step {currentStep} of {STEPS.length} / Qajeelfannaa {currentStep} kiyya {STEPS.length} keessaa</span>
   293→          <button type="button" onClick={resetForm} className="flex items-center gap-1 text-red-500 hover:text-red-700">
   294→            <RotateCcw className="h-3 w-3" />
   295→            <span className="hidden sm:inline">Reset / Balleessuu</span><span className="sm:hidden">Reset</span>
   296→          </button>
   297→        </div>
   298→      </div>
   299→
   300→      <form onSubmit={handleSubmit}>
   301→        <Card className="min-h-[300px] sm:min-h-[360px]">
   302→          {/* Step 1: Enterprise */}
   303→          {currentStep === 1 && (
   304→            <>
   305→              <CardHeader className="pb-2 sm:pb-3">
   306→                <CardTitle className="flex items-center gap-1.5 text-base sm:gap-2 sm:text-lg">
   307→                  <Store className="h-4 w-4 shrink-0 text-amber-600 sm:h-5 sm:w-5" />
   308→                  <span className="leading-tight">Odeeffannoo Midhaalee<br /><span className="text-sm font-normal text-muted-foreground sm:text-base">Small enterprise information</span></span>
   309→                </CardTitle>
   310→                <CardDescription className="text-xs sm:text-sm">Basic details about the small enterprise (MSE)</CardDescription>
   311→              </CardHeader>
   312→              <CardContent className="space-y-3 sm:space-y-4">
   313→                <div className="space-y-1.5 sm:space-y-2">
   314→                  <Label htmlFor="enterpriseName" className="text-xs sm:text-sm">Enterprise Name <span className="text-red-500">*</span></Label>
   315→                  <Input id="enterpriseName" name="enterpriseName" placeholder="e.g., TIGIST TAILORING WORKSHOP" value={formData.enterpriseName} onChange={handleChange} required className="text-sm sm:text-base" />
   316→                </div>
   317→                <div className="space-y-1.5 sm:space-y-2">
   318→                  <Label htmlFor="enterpriseType" className="text-xs sm:text-sm">Enterprise Type / Gosa Midhaalee <span className="text-red-500">*</span></Label>
   319→                  <Select value={formData.enterpriseType} onValueChange={(v) => handleSelectChange('enterpriseType', v)}>
   320→                    <SelectTrigger id="enterpriseType" className="text-sm sm:text-base"><SelectValue placeholder="Select / Filadhu" /></SelectTrigger>
   321→                    <SelectContent className="max-h-64">{SMALL_ENTERPRISE_TYPES.map((t) => (<SelectItem key={t} value={t}>{t}</SelectItem>))}</SelectContent>
   322→                  </Select>
   323→                </div>
   324→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 sm:gap-4">
   325→                  <div className="space-y-1.5 sm:space-y-2">
   326→                    <Label htmlFor="subCity" className="text-xs sm:text-sm">Sub-City<br /><span className="text-xs font-normal text-muted-foreground">Kuttaa Maggalaa</span> <span className="text-red-500">*</span></Label>
   327→                    <Select value={formData.subCity} onValueChange={handleSubCityChange}>
   328→                      <SelectTrigger id="subCity" className="text-sm sm:text-base"><SelectValue placeholder="Select / Filadhu" /></SelectTrigger>
   329→                      <SelectContent>{subCities.map((sc) => (<SelectItem key={sc.name} value={sc.name}>{sc.name}</SelectItem>))}</SelectContent>
   330→                    </Select>
   331→                  </div>
   332→                  <div className="space-y-1.5 sm:space-y-2">
   333→                    <Label htmlFor="area" className="text-xs sm:text-sm">Werreda <span className="text-red-500">*</span></Label>
   334→                    <Select value={formData.area} onValueChange={(v) => handleSelectChange('area', v)} disabled={!selectedSubCity}>
   335→                      <SelectTrigger id="area" className="text-sm sm:text-base"><SelectValue placeholder={selectedSubCity ? 'Select / Filadhu' : 'Select sub-city first'} /></SelectTrigger>
   336→                      <SelectContent>{areas.map((a) => (<SelectItem key={a} value={a}>{a}</SelectItem>))}</SelectContent>
   337→                    </Select>
   338→                  </div>
   339→                </div>
   340→                <div className="space-y-1.5 sm:space-y-2">
   341→                  <Label htmlFor="specificAddress" className="text-xs sm:text-sm">Specific Address<br /><span className="text-xs font-normal text-muted-foreground">Baka Addaa</span> <span className="text-red-500">*</span></Label>
   342→                  <Input id="specificAddress" name="specificAddress" placeholder="Landmark, street, plot number" value={formData.specificAddress} onChange={handleChange} required className="text-sm sm:text-base" />
   343→                </div>
   344→              </CardContent>
   345→            </>
   346→          )}
   347→
   348→          {/* Step 2: Capacity & Support */}
   349→          {currentStep === 2 && (
   350→            <>
   351→              <CardHeader className="pb-2 sm:pb-3">
   352→                <CardTitle className="flex items-center gap-1.5 text-base sm:gap-2 sm:text-lg">
   353→                  <FileText className="h-4 w-4 shrink-0 text-amber-600 sm:h-5 sm:w-5" />
   354→                  <span className="leading-tight">Dandeettii fi Deeggarsa<br /><span className="text-sm font-normal text-muted-foreground sm:text-base">Capacity &amp; support</span></span>
   355→                </CardTitle>
   356→                <CardDescription className="text-xs sm:text-sm">License, ownership, employees, capital and support</CardDescription>
   357→              </CardHeader>
   358→              <CardContent className="space-y-3 sm:space-y-4">
   359→                <div className="space-y-1.5 sm:space-y-2">
   360→                  <Label htmlFor="licenseNumber" className="text-xs sm:text-sm">License Number<br /><span className="text-xs font-normal text-muted-foreground">Lakofsaa Eyyeema</span></Label>
   361→                  <Input id="licenseNumber" name="licenseNumber" placeholder="e.g., MSE/2024/012" value={formData.licenseNumber} onChange={handleChange} className="text-sm sm:text-base" />
   362→                </div>
   363→                <div className="space-y-1.5 sm:space-y-2">
   364→                  <Label className="text-xs sm:text-sm">Ownership Type / Gosa Abbaa Qabuu <span className="text-red-500">*</span></Label>
   365→                  <RadioGroup value={formData.ownershipType} onValueChange={(v) => handleSelectChange('ownershipType', v)} className="grid grid-cols-1 gap-2 sm:grid-cols-2">
   366→                    {SMALL_ENTERPRISE_OWNERSHIP_TYPES.map((t) => (
   367→                      <div key={t} className="flex items-center space-x-2 rounded-lg border p-2.5 has-data-[state=checked]:border-amber-500 has-data-[state=checked]:bg-amber-50">
   368→                        <RadioGroupItem value={t} id={`mse-ownership-${t}`} />
   369→                        <Label htmlFor={`mse-ownership-${t}`} className="cursor-pointer text-xs sm:text-sm font-normal">{t}</Label>
   370→                      </div>
   371→                    ))}
   372→                  </RadioGroup>
   373→                </div>
   374→                <div className="space-y-1.5 sm:space-y-2">
   375→                  <Label htmlFor="employeeCount" className="text-xs sm:text-sm">Employee Count<br /><span className="text-xs font-normal text-muted-foreground">Lakkoofsa Hojjetaa</span> <span className="text-red-500">*</span></Label>
   376→                  <Input id="employeeCount" name="employeeCount" type="number" min="0" placeholder="e.g., 8" value={formData.employeeCount} onChange={handleChange} required className="text-sm sm:text-base" />
   377→                </div>
   378→                {/* Employees by sex (male / female) */}
   379→                <div className="rounded-lg border border-amber-100 bg-amber-50/50 p-3 space-y-2.5">
   380→                  <div className="flex items-center justify-between gap-2">
   381→                    <Label className="text-xs sm:text-sm">Employees by Sex<br /><span className="text-xs font-normal text-muted-foreground">Hojjetaa Dhiira fi Dubartii</span></Label>
   382→                    <span className="shrink-0 rounded-full bg-amber-100 px-2 py-0.5 text-[9px] font-medium text-amber-700 sm:text-[10px]">Optional / Filannoo</span>
   383→                  </div>
   384→                  <div className="grid grid-cols-2 gap-3">
   385→                    <div className="space-y-1.5">
   386→                      <Label htmlFor="maleEmployees" className="text-xs">Male / Dhiira</Label>
   387→                      <Input id="maleEmployees" name="maleEmployees" type="number" min="0" placeholder="e.g., 3" value={formData.maleEmployees} onChange={handleSexChange} className="text-sm sm:text-base" />
   388→                    </div>
   389→                    <div className="space-y-1.5">
