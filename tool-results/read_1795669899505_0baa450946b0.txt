     1→/**
     2→ * Generic registry of the admin-managed survey questionnaires — one per data
     3→ * collection module (Hotel & Guest House, Health, Education, Agriculture,
     4→ * Small Enterprise). The Industry module keeps its own registry
     5→ * (industry-survey-fields.ts) with identical semantics.
     6→ *
     7→ * Admins manage each questionnaire from the module's "Manage" workspace (and
     8→ * Administration → Settings → Survey): hide/disable optional questions, flip
     9→ * required flags, edit the bilingual labels and reorder questions inside
    10→ * their step. The wizard form, its step validation, the review list and the
    11→ * server-side POST validation all follow this config. Core identification
    12→ * fields (name, type, location, headline count, contact) are locked — records
    13→ * would lose their backbone without them. Male/female breakdowns count as ONE
    14→ * question ("…Sex" pseudo-keys), like the Industry "employeeSex" precedent.
    15→ *
    16→ * Storage: AppSetting "<sectorId>SurveyFields" — an ordered array of
    17→ * { key, labelEn, labelOr, enabled, required }. Order within a wizard step
    18→ * follows the saved array order; locked fields keep their registry position.
    19→ * Hidden questions store empty/default values for new records.
    20→ */
    21→
    22→export type SurveyInput =
    23→  | 'text'
    24→  | 'number'
    25→  | 'select'
    26→  | 'select-multi'
    27→  | 'radio-yesno'
    28→  | 'sex-pair'
    29→  | 'status-select';
    30→
    31→export interface SurveyFieldDef {
    32→  /** Payload field name (matches the model column / form field). */
    33→  key: string;
    34→  /** Wizard step the question sits on (forms use 1..3 + a fixed review step). */
    35→  step: 1 | 2 | 3;
    36→  /** Default English label. */
    37→  labelEn: string;
    38→  /** Default Afaan Oromoo sub-label. */
    39→  labelOr: string;
    40→  /** How the form renders the question. */
    41→  input: SurveyInput;
    42→  /** Whether it is required by default. */
    43→  required: boolean;
    44→  /** Cannot be disabled (core identification fields). */
    45→  locked: boolean;
    46→  /** Required flag is fixed (cannot be toggled in Settings). */
    47→  requiredFixed: boolean;
    48→}
    49→
    50→export interface SurveyFieldConfig {
    51→  key: string;
    52→  labelEn: string;
    53→  labelOr: string;
    54→  enabled: boolean;
    55→  required: boolean;
    56→  /** Admin-added custom question (SME Manage tab) — carries its own input/step/options. */
    57→  custom?: boolean;
    58→  /** Custom questions only: wizard step the question sits on (1..3). */
    59→  step?: 1 | 2 | 3;
    60→  /** Custom questions only: how the form renders the question. */
    61→  input?: CustomSurveyInput;
    62→  /** Custom questions only: choices for a select input. */
    63→  options?: string[];
    64→}
    65→
    66→/** Input types available to admin-added custom questions. */
    67→export type CustomSurveyInput = 'text' | 'number' | 'textarea' | 'select' | 'radio-yesno';
    68→
    69→export const CUSTOM_SURVEY_INPUTS: { value: CustomSurveyInput; label: string }[] = [
    70→  { value: 'text', label: 'Short text' },
    71→  { value: 'number', label: 'Number' },
    72→  { value: 'textarea', label: 'Paragraph' },
    73→  { value: 'select', label: 'Dropdown (single choice)' },
    74→  { value: 'radio-yesno', label: 'Yes / No' },
    75→];

    77→/** Prefix of admin-added custom question keys. */
    78→export const CUSTOM_SURVEY_KEY_PREFIX = 'custom_';
    79→
    80→export function isCustomSurveyKey(key: unknown): key is string {
    81→  return typeof key === 'string' && key.startsWith(CUSTOM_SURVEY_KEY_PREFIX) && key.length > CUSTOM_SURVEY_KEY_PREFIX.length;
    82→}
    83→
    84→/** Builds a unique key for a new custom question from its English label. */
    85→export function makeCustomSurveyKey(labelEn: string, existingKeys: string[]): string {
    86→  const base = CUSTOM_SURVEY_KEY_PREFIX + (
    87→    labelEn.toLowerCase().replace(/[^a-z0-9]+/g, '_').replace(/^_+|_+$/g, '').slice(0, 40) || 'question'
    88→  );
    89→  let key = base;
    90→  let i = 2;
    91→  while (existingKeys.includes(key)) {
    92→    key = `${base}_${i}`;
    93→    i += 1;
    94→  }
    95→  return key;
    96→}
    97→
    98→/**
    99→ * Keeps only the answers of enabled custom questions, as plain strings.
   100→ * Unknown / disabled keys are dropped; values are trimmed and capped.
   101→ */
   102→export function sanitizeCustomAnswers(
   103→  fields: SurveyFieldConfig[],
   104→  raw: unknown
   105→): Record<string, string> {
   106→  const customKeys = new Set(fields.filter((f) => f.custom && f.enabled).map((f) => f.key));
   107→  const out: Record<string, string> = {};
   108→  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return out;
   109→  for (const [k, v] of Object.entries(raw as Record<string, unknown>)) {
   110→    if (!customKeys.has(k)) continue;
   111→    if (v == null) continue;
   112→    const s = String(v).trim().slice(0, 500);
   113→    if (s) out[k] = s;
   114→  }
   115→  return out;
   116→}

  1058→/**
  1059→ * Validates an admin-managed questionnaire for a sector. Keeps the saved order
  1060→ * for known keys (per step, since steps are fixed), drops unknown keys and
  1061→ * duplicates, forces locked fields enabled and fixed-required fields required,
  1062→ * trims labels (falls back to the registry default when empty). Returns null
  1063→ * when nothing valid remains (the caller rejects the save).
  1064→ */
  1065→export function normalizeSectorSurveyFields(
  1066→  sector: SurveySectorId,
  1067→  raw: unknown
  1068→): SurveyFieldConfig[] | null {
  1069→  if (!Array.isArray(raw)) return null;
  1070→  const registry = SURVEY_SECTORS[sector].fields;
  1071→  const seen = new Set<string>();
  1072→  const out: SurveyFieldConfig[] = [];
  1073→  for (const item of raw) {
  1074→    const key = (item as { key?: unknown })?.key;
  1075→    // Admin-added custom questions ride along with their own shape.
  1076→    if (isCustomSurveyKey(key)) {
  1077→      if (seen.has(key)) continue;
  1078→      seen.add(key);
  1079→      const it = item as {
  1080→        labelEn?: unknown; labelOr?: unknown; enabled?: unknown;
  1081→        required?: unknown; step?: unknown; input?: unknown; options?: unknown;
  1082→      };
  1083→      const step = it.step === 1 || it.step === 2 || it.step === 3 ? it.step : 3;
  1084→      const input = CUSTOM_SURVEY_INPUTS.some((i) => i.value === it.input)
  1085→        ? (it.input as CustomSurveyInput)
  1086→        : 'text';
  1087→      const options = input === 'select' && Array.isArray(it.options)
  1088→        ? it.options
  1089→            .map((o) => (typeof o === 'string' ? o.trim().slice(0, 80) : ''))
  1090→            .filter((o) => o.length > 0)
  1091→            .slice(0, 30)
  1092→        : undefined;
  1093→      out.push({
  1094→        key,
  1095→        labelEn: typeof it.labelEn === 'string' && it.labelEn.trim()
  1096→          ? it.labelEn.trim().slice(0, 120)
  1097→          : 'Custom question',
  1098→        labelOr: typeof it.labelOr === 'string' ? it.labelOr.trim().slice(0, 120) : '',
  1099→        enabled: it.enabled !== false,
  1100→        required: it.required === true,
  1101→        custom: true,
  1102→        step,
  1103→        input,
  1104→        ...(options ? { options } : {}),
  1105→      });
  1106→      continue;
  1107→    }
  1108→    if (!isSectorSurveyFieldKey(sector, key) || seen.has(key)) continue;
  1109→    seen.add(key);
  1110→    const def = registry.find((f) => f.key === key)!;
  1111→    const labelEn =
  1112→      typeof (item as { labelEn?: unknown })?.labelEn === 'string' &&
  1113→      (item as { labelEn?: string }).labelEn!.trim()
  1114→        ? (item as { labelEn?: string }).labelEn!.trim().slice(0, 120)
  1115→        : def.labelEn;
  1116→    const labelOr =
  1117→      typeof (item as { labelOr?: unknown })?.labelOr === 'string' &&
  1118→      (item as { labelOr?: string }).labelOr!.trim()
  1119→        ? (item as { labelOr?: string }).labelOr!.trim().slice(0, 120)
  1120→        : def.labelOr;
  1121→    out.push({
  1122→      key,
  1123→      labelEn,
  1124→      labelOr,
  1125→      enabled: def.locked ? true : (item as { enabled?: unknown })?.enabled !== false,
  1126→      required: def.requiredFixed ? true : (item as { required?: unknown })?.required === true,
  1127→    });
  1128→  }
  1129→  // Every registry field must be present — append any the admin list missed so
  1130→  // a stale config can never drop a locked field.
  1131→  for (const def of registry) {
  1132→    if (!seen.has(def.key)) {
  1133→      out.push({
  1134→        key: def.key,
  1135→        labelEn: def.labelEn,
  1136→        labelOr: def.labelOr,
  1137→        enabled: true,
  1138→        required: def.required,
  1139→      });
  1140→    }
  1141→  }
  1142→  return out.length > 0 ? out : null;
  1143→}
  1144→
  1145→/**
  1146→ * Merges a saved config with the registry — the read-side equivalent of
  1147→ * normalizeSectorSurveyFields for configs persisted earlier. Falls back to
  1148→ * the defaults when nothing valid was saved.
  1149→ */
  1150→export function mergeSectorSurveyFields(
  1151→  sector: SurveySectorId,
  1152→  saved: unknown
  1153→): SurveyFieldConfig[] {
  1154→  return normalizeSectorSurveyFields(sector, saved) ?? sectorSurveyFieldsDefault(sector);
  1155→}

  1169→  fields: SurveyFieldConfig[],

  1177→/** Whether a shown field must be filled before submission. */
  1178→export function sectorSurveyFieldRequired(
  1179→  sector: SurveySectorId,
  1180→  fields: SurveyFieldConfig[],
  1181→  key: string
