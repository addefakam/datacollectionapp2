   128→export default function HealthList({ refreshTrigger }: DataListProps) {
   129→  const { subCities, areasFor } = useSettings();
   130→  const { toast } = useToast();
   131→  const [data, setData] = useState<HealthCenter[]>([]);
   132→  const [loading, setLoading] = useState(true);
   133→  const [search, setSearch] = useState('');
   134→  const [filterSubCity, setFilterSubCity] = useState('');
   135→  const [filterArea, setFilterArea] = useState('');
   136→  const [filterType, setFilterType] = useState('');
   137→  const [areas, setAreas] = useState<string[]>([]);
   138→  const [page, setPage] = useState(1);
   139→  const [totalPages, setTotalPages] = useState(1);
   140→  const [total, setTotal] = useState(0);
   141→  const [deleteId, setDeleteId] = useState<string | null>(null);
   142→  const [viewItem, setViewItem] = useState<HealthCenter | null>(null);
   143→  const [editItem, setEditItem] = useState<HealthCenter | null>(null);
   144→  const [editForm, setEditForm] = useState<EditForm | null>(null);
   145→  const [editAreas, setEditAreas] = useState<string[]>([]);
   146→  const [saving, setSaving] = useState(false);
   147→  const [deleting, setDeleting] = useState(false);
   148→  const [exporting, setExporting] = useState(false);
   149→  const [importing, setImporting] = useState(false);
   150→  const [showImport, setShowImport] = useState(false);
   151→  const [importResult, setImportResult] = useState<{success: number, failed: number, errors: string[]} | null>(null);
   152→  const fileInputRef = useRef<HTMLInputElement>(null);
   153→
   154→  const fetchData = useCallback(async () => {
   155→    setLoading(true);
   156→    try {
   157→      const params = new URLSearchParams();
   158→      if (search) params.set('search', search);
   159→      if (filterSubCity) params.set('subCity', filterSubCity);
   160→      if (filterArea) params.set('area', filterArea);
   161→      if (filterType) params.set('facilityType', filterType);
   162→      params.set('page', String(page));
   163→      params.set('limit', String(PAGE_SIZE));
   164→
   165→      const res = await fetch(`/api/healthcenters?${params}`);
   166→      if (!res.ok) throw new Error('Failed to fetch');
   167→      const json = await res.json();
   168→      setData(json.data);
   169→      setTotal(json.total);
   170→      setTotalPages(json.totalPages);
   171→    } catch {
   172→      toast({
   173→        title: 'Error',
   174→        description: 'Failed to load data',
   175→        variant: 'destructive',
   176→      });
   177→    } finally {
   178→      setLoading(false);
   179→    }
   180→  }, [search, filterSubCity, filterArea, filterType, page, toast]);
   181→
   182→  useEffect(() => {
   183→    fetchData();
   184→  }, [fetchData, refreshTrigger]);
   185→
   186→  const handleSubCityFilterChange = (value: string) => {
   187→    setFilterSubCity(value);
   188→    setFilterArea('');
   189→    setPage(1);
   190→    setAreas(value ? areasFor(value) : []);
   191→  };
   192→
   193→  const handleDelete = async () => {
   194→    if (!deleteId) return;
   195→    setDeleting(true);
   196→    try {
   197→      const res = await fetch(`/api/healthcenters/${deleteId}`, {
   198→        method: 'DELETE',
   199→      });
   200→      if (!res.ok) throw new Error('Delete failed');
   201→      toast({ title: 'Deleted', description: 'Record removed successfully.' });
   202→      fetchData();
   203→    } catch {
   204→      toast({
   205→        title: 'Delete Failed',
   206→        description: 'Could not delete the record.',
   207→        variant: 'destructive',
   208→      });

   210→      setDeleting(false);
   211→      setDeleteId(null);
   212→    }
   213→  };
   214→

   216→    setEditItem(item);
   217→    setEditForm({
   218→      facilityName: item.facilityName,
   219→      facilityType: item.facilityType,
   220→      ownershipType: item.ownershipType,
   221→      subCity: item.subCity,
   222→      area: item.area,
   223→      specificAddress: item.specificAddress,
   224→      licenseNumber: item.licenseNumber || '',
   225→      staffCount: String(item.staffCount),
   226→      bedCount: String(item.bedCount),
   227→      hasLaboratory: item.hasLaboratory ? 'yes' : 'no',
   228→      hasPharmacy: item.hasPharmacy ? 'yes' : 'no',
   229→      hasMaternity: item.hasMaternity ? 'yes' : 'no',
   230→      hasEmergency: item.hasEmergency ? 'yes' : 'no',

   232→      contactPhone: item.contactPhone,

   238→    if (!editForm) return;
   239→    setEditAreas(areasFor(value));
   240→    setEditForm({ ...editForm, subCity: value, area: '' });
   241→  };
   242→

   244→    if (!editItem || !editForm) return;
   245→    setSaving(true);
   246→    try {
   247→      const res = await fetch(`/api/healthcenters/${editItem.id}`, {
   248→        method: 'PUT',
   249→        headers: { 'Content-Type': 'application/json' },
   250→        body: JSON.stringify(editForm),

   252→      if (!res.ok) {
   253→        const err = await res.json().catch(() => ({}));
   254→        throw new Error(err.error || 'Update failed');

   256→      toast({ title: 'Saved / Nuusame', description: 'Record updated successfully' });
   257→      setEditItem(null);
   258→      setEditForm(null);
   259→      fetchData();

   261→      toast({ title: 'Error', description: error instanceof Error ? error.message : 'Failed to save', variant: 'destructive' });
   262→    } finally {
   263→      setSaving(false);
   264→    }
   265→  };
   266→

   268→    setExporting(true);

   273→      if (filterType) params.set('facilityType', filterType);

   277→        throw new Error(err.error || 'Export failed');

   289→    } catch (error) {
   290→      toast({ title: 'Export Failed', description: error instanceof Error ? error.message : 'Failed to export', variant: 'destructive' });
   291→    } finally {

   313→  };
   314→
   315→  const actionButtons = (item: HealthCenter) => (

   322→      </Button>
   323→      <Button variant="ghost" size="icon" className="h-7 w-7 text-red-500 hover:text-red-600 hover:bg-red-50" onClick={() => setDeleteId(item.id)} title="Delete">
   324→        <Trash2 className="h-3.5 w-3.5" />
   325→      </Button>
   326→    </>
   327→  );
   328→
   329→  const serviceChips = (item: HealthCenter, size: 'sm' | 'lg' = 'sm') => {
   330→    const svc = [
   331→      { ok: item.hasLaboratory, short: 'Lab', icon: FlaskConical },
   332→      { ok: item.hasPharmacy, short: 'Pharm', icon: Pill },
   333→      { ok: item.hasMaternity, short: 'Mat', icon: Baby },
   334→      { ok: item.hasEmergency, short: '24/7', icon: Siren },
   335→    ];
   336→    const cls = size === 'sm'
   337→      ? 'inline-flex items-center gap-0.5 rounded-full px-1.5 py-0.5 text-[9px] font-semibold'
   338→      : 'inline-flex items-center gap-0.5 rounded-full px-2 py-0.5 text-xs font-semibold';
   339→    return svc.map((s) => (
   340→      <span key={s.short} className={`${cls} ${s.ok ? 'bg-green-100 text-green-700' : 'bg-slate-100 text-slate-400'}`}>
   341→        {s.ok ? <Check className="h-2.5 w-2.5" /> : <X className="h-2.5 w-2.5" />} {s.short}
   342→      </span>
   343→    ));
   344→  };
   345→
   346→  const startIndex = (page - 1) * PAGE_SIZE;
   347→
   348→  return (
   349→    <div className="space-y-3 sm:space-y-4">
   350→      <Card>
   351→        <CardHeader className="pb-2 sm:pb-3">
   352→          <CardTitle className="flex items-center justify-between gap-2 text-base sm:text-lg">
   353→            <span className="flex items-center gap-2">
   354→              <Stethoscope className="h-5 w-5 text-sky-600" />
   355→              Health Records / Galmee Fayyummaa
   356→            </span>
   357→            <Badge variant="secondary" className="shrink-0 bg-sky-100 text-sky-700 text-xs sm:text-sm">
   358→              {total} record{total !== 1 ? 's' : ''}
   359→            </Badge>
   360→          </CardTitle>
   361→          <CardDescription className="text-xs sm:text-sm">View and manage all collected health facility survey data</CardDescription>
   362→        </CardHeader>
   363→        <CardContent className="space-y-2.5 sm:space-y-3">
   364→          {/* Search */}
   365→          <div className="relative">
   366→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   367→            <Input
   368→              placeholder="Search name, address, license, contact..."
   369→              value={search}
   370→              onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   371→              className="pl-9 text-sm sm:text-base"
   372→            />
   373→          </div>
   374→
   375→          {/* Filters */}
   376→          <div className="grid grid-cols-1 gap-2 sm:grid-cols-3 sm:gap-2">
   377→            <Select value={filterType || undefined} onValueChange={(v) => { setFilterType(v === 'all' ? '' : v); setPage(1); }}>
   378→              <SelectTrigger className="text-xs sm:text-sm"><SelectValue placeholder="All facility types" /></SelectTrigger>

   389→              </SelectContent>
   390→            </Select>
   391→            <Select value={filterArea || undefined} onValueChange={(v) => { setFilterArea(v === 'all' ? '' : v); setPage(1); }} disabled={!filterSubCity}>
   392→              <SelectTrigger className="text-xs sm:text-sm"><SelectValue placeholder={filterSubCity ? 'All woredas' : 'Select sub-city first'} /></SelectTrigger>
   393→              <SelectContent>
   394→                <SelectItem value="all">All woredas</SelectItem>
   395→                {areas.map((a) => (<SelectItem key={a} value={a}>{a}</SelectItem>))}
   396→              </SelectContent>
   397→            </Select>
   398→          </div>
   399→
   400→          {/* Export / Import buttons */}
   401→          <div className="flex gap-2">
   402→            <Button variant="outline" size="sm" onClick={() => handleExport('csv')} disabled={exporting} className="flex-1 text-xs sm:text-sm">
   403→              <Download className="h-3.5 w-3.5" /> CSV

   405→            <Button variant="outline" size="sm" onClick={() => handleExport('excel')} disabled={exporting} className="flex-1 text-xs sm:text-sm">
   406→              <FileSpreadsheet className="h-3.5 w-3.5" /> Excel

   409→              <Upload className="h-3.5 w-3.5" /> Import

   417→
   418→      {/* Records List */}
   419→      {loading ? (
   420→        <div className="space-y-2">
   421→          {[1, 2, 3, 4, 5].map((i) => <Skeleton key={i} className="h-20 w-full" />)}
   422→        </div>
   423→      ) : data.length === 0 ? (
   424→        <Card>
   425→          <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   426→            <Stethoscope className="mb-3 h-10 w-10 text-muted-foreground/40" />
   427→            <p className="text-sm font-medium text-muted-foreground">No health facility records found</p>
   428→            <p className="mt-1 text-xs text-muted-foreground/70">Collect a survey, import Excel, or adjust your filters</p>
   429→          </CardContent>
   430→        </Card>
   431→      ) : (
   432→        <>
   433→          {/* Desktop: full data table */}
   434→          <Card className="hidden md:block overflow-hidden">
   435→            <CardContent className="px-0 py-0">
   436→              <table className="w-full text-xs">
   437→                <thead>
   438→                  <tr className="border-b bg-gradient-to-r from-slate-50 via-sky-50/60 to-slate-50">
   439→                    <th className="w-10 px-2.5 py-2 text-center font-semibold text-[10px] uppercase tracking-wider text-sky-900/60">#</th>

   443→                    <th className="px-2.5 py-2 text-left font-semibold text-[10px] uppercase tracking-wider text-sky-900/60">Werreda</th>
   444→                    <th className="px-2.5 py-2 text-left font-semibold text-[10px] uppercase tracking-wider text-sky-900/60">License No</th>
