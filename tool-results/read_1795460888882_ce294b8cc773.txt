     1→'use client';
     2→
     3→import { useState } from 'react';
     4→import { Button } from '@/components/ui/button';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import {
     8→  Select,
     9→  SelectContent,
    10→  SelectItem,
    11→  SelectTrigger,
    12→  SelectValue,
    13→} from '@/components/ui/select';
    14→import {
    15→  Card,
    16→  CardContent,
    17→  CardDescription,
    18→  CardHeader,
    19→  CardTitle,
    20→} from '@/components/ui/card';
    21→import { useToast } from '@/hooks/use-toast';
    22→import {
    23→  GraduationCap,
    24→  Loader2,
    25→  RotateCcw,
    26→  ChevronLeft,
    27→  ChevronRight,
    28→  Check,
    29→  UserCircle,
    30→  Phone,
    31→  FileText,
    32→  Eye,
    33→} from 'lucide-react';
    34→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group';
    35→import { EDUCATION_INSTITUTION_TYPES, EDUCATION_OWNERSHIP_TYPES } from '@/lib/location-data';
    36→import { useSettings } from '@/components/SettingsProvider';
    37→
    38→interface EducationSurveyFormProps {
    39→  onSubmit: () => void;
    40→  surveyorName: string;
    41→  surveyorId: string;
    42→  isOnline?: boolean;
    43→  onOfflineSave?: (data: Record<string, unknown>) => Promise<boolean>;
    44→}
    45→
    46→const STEPS = [
    47→  { id: 1, title: 'Institution', titleOr: 'Mana Barumsaa', icon: GraduationCap },
    48→  { id: 2, title: 'Capacity', titleOr: 'Dandeettii', icon: FileText },
    49→  { id: 3, title: 'Contact', titleOr: 'Quunnamtii', icon: Phone },
    50→  { id: 4, title: 'Review', titleOr: 'Mirkanaa', icon: Eye },
    51→];
    52→
    53→const SERVICE_FIELDS: { name: string; label: string; labelOr: string; hint: string }[] = [
    54→  { name: 'hasLibrary', label: 'Library service', labelOr: 'Mana Maxxantaa', hint: 'Library or reading room' },
    55→  { name: 'hasLaboratory', label: 'Laboratory service', labelOr: 'Laboratoire', hint: 'Science laboratory' },
    56→  { name: 'hasComputerLab', label: 'Computer lab', labelOr: 'Kompiyuutaraa', hint: 'Computer or ICT lab' },
    57→  { name: 'hasFeedingProgram', label: 'Feeding program', labelOr: 'Soorataa', hint: 'School meals program' },
    58→];
    59→
    60→export default function EducationSurveyForm({
    61→  onSubmit,
    62→  surveyorName,
    63→  surveyorId,
    64→  isOnline = true,
    65→  onOfflineSave,
    66→}: EducationSurveyFormProps) {
    67→  const { toast } = useToast();
    68→  const { subCities, areasFor } = useSettings();
    69→  const [isSubmitting, setIsSubmitting] = useState(false);
    70→  const [currentStep, setCurrentStep] = useState(1);
    71→  const [selectedSubCity, setSelectedSubCity] = useState('');
    72→  const [areas, setAreas] = useState<string[]>([]);
    73→  const [dataConfirmed, setDataConfirmed] = useState('');
    74→
    75→  const emptyForm = {
    76→    institutionName: '',
    77→    institutionType: '',
    78→    ownershipType: '',
    79→    subCity: '',
    80→    area: '',
    81→    specificAddress: '',
    82→    licenseNumber: '',
    83→    studentCount: '',
    84→    teacherCount: '',
    85→    classroomCount: '',
    86→    hasLibrary: 'no',
    87→    hasLaboratory: 'no',
    88→    hasComputerLab: 'no',
    89→    hasFeedingProgram: 'no',
    90→    contactName: '',
    91→    contactPhone: '',
    92→  };
    93→
    94→  const [formData, setFormData] = useState(emptyForm);
    95→
    96→  const handleSubCityChange = (value: string) => {
    97→    setSelectedSubCity(value);
    98→    setAreas(areasFor(value));
    99→    setFormData((prev) => ({ ...prev, subCity: value, area: '' }));
   100→  };
   101→
   102→  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
   103→    const { name, value } = e.target;
   104→    setFormData((prev) => ({ ...prev, [name]: value }));
   105→  };
   106→
   107→  const handleSelectChange = (name: string, value: string) => {
   108→    setFormData((prev) => ({ ...prev, [name]: value }));
   109→  };
   110→
   111→  const validateStep = (step: number): boolean => {
   112→    switch (step) {
   113→      case 1:
   114→        if (!formData.institutionName || !formData.institutionType || !formData.subCity || !formData.area || !formData.specificAddress) {
   115→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: 'All fields in this step are required / Galii hundinuu barbaachisaadha', variant: 'destructive' });
   116→          return false;
   117→        }
   118→        return true;
   119→      case 2:
   120→        if (!formData.ownershipType) {
   121→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: 'Ownership type is required / Gosa abbaa qabuu barbaachisaadha', variant: 'destructive' });
   122→          return false;
   123→        }
   124→        if (!formData.studentCount || Number(formData.studentCount) < 0) {
   125→          toast({ title: 'Invalid Student Count / Lakkoofsa Barataa Dogoggora', description: 'Enter a valid student count / Lakkoofsa barataa sirrii galchi', variant: 'destructive' });
   126→          return false;
   127→        }
   128→        if (!formData.teacherCount || Number(formData.teacherCount) < 0) {
   129→          toast({ title: 'Invalid Teacher Count / Lakkoofsa Barsiisaa Dogoggora', description: 'Enter a valid teacher count / Lakkoofsa barsiisaa sirrii galchi', variant: 'destructive' });
   130→          return false;
   131→        }
   132→        if (!formData.classroomCount || Number(formData.classroomCount) < 0) {
   133→          toast({ title: 'Invalid Classroom Count', description: 'Enter a valid classroom count (0 or more)', variant: 'destructive' });
   134→          return false;
   135→        }
   136→        return true;
   137→      case 3:
   138→        if (!formData.contactName || !formData.contactPhone) {
   139→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: 'Contact person and phone are required / Maqaa fi bilbila quunnamtii barbaachisa', variant: 'destructive' });
   140→          return false;
   141→        }
   142→        {
   143→          const phoneClean = formData.contactPhone.replace(/[\s-]/g, '');
   144→          const ethPhoneRegex = /^(\+251|251|0)?(9|7)\d{8}$/;
   145→          if (!ethPhoneRegex.test(phoneClean)) {
   146→            toast({ title: 'Invalid Phone / Bilbila Dogoggora', description: 'Enter valid Ethiopian number (e.g., +251 91 234 5678)', variant: 'destructive' });
   147→            return false;
   148→          }
   149→        }
   150→        return true;
   151→      case 4:
   152→        if (dataConfirmed !== 'yes') {
   153→          toast({ title: 'Confirmation Required / Mirkanaa\u2019uu Barbaachisaa', description: 'Please confirm data is true and authenticated / Daataan dhugaa fi mirkanaa\u2019eef mirkanaadhu', variant: 'destructive' });
   154→          return false;
   155→        }
   156→        return true;
   157→      default:
   158→        return true;
   159→    }
   160→  };
   161→
   162→  const nextStep = () => { if (validateStep(currentStep)) setCurrentStep((s) => Math.min(s + 1, STEPS.length)); };
   163→  const prevStep = () => { setCurrentStep((s) => Math.max(s - 1, 1)); };
   164→
   165→  const resetForm = () => {
   166→    setFormData(emptyForm);
   167→    setSelectedSubCity('');
   168→    setAreas([]);
   169→    setDataConfirmed('');
   170→    setCurrentStep(1);
   171→  };
   172→
   173→  const handleSubmit = async (e: React.FormEvent) => {
   174→    e.preventDefault();
   175→    if (!validateStep(currentStep)) return;
   176→    const payload = {
   177→      ...formData,
   178→      studentCount: Number(formData.studentCount || 0),
   179→      teacherCount: Number(formData.teacherCount || 0),
   180→      classroomCount: Number(formData.classroomCount || 0),
   181→      hasLibrary: formData.hasLibrary === 'yes',
   182→      hasLaboratory: formData.hasLaboratory === 'yes',
   183→      hasComputerLab: formData.hasComputerLab === 'yes',
   184→      hasFeedingProgram: formData.hasFeedingProgram === 'yes',
   185→      surveyorName,
   186→      surveyorId,
   187→    };
   188→    setIsSubmitting(true);
   189→    if (isOnline) {
   190→      try {
   191→        const response = await fetch('/api/educationcenters', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
   192→        if (!response.ok) { const err = await response.json(); throw new Error(err.error || 'Failed to submit'); }
   193→        toast({ title: 'Survey Submitted! / Galma Qabame!', description: `${formData.institutionName} recorded successfully / galmeen safiisan argame` });
   194→        resetForm();
   195→        onSubmit();
   196→      } catch (error) {
   197→        if (onOfflineSave) {
   198→          const saved = await onOfflineSave(payload);
   199→          if (saved) { toast({ title: 'Saved Offline / Offliin Kuusame', description: 'Will sync when connected / Netiirkii qunnamaa waanin dhiheenya' }); resetForm(); onSubmit(); }
   200→          else { toast({ title: 'Submission Failed / Galmeen Hin Dhufne', description: error instanceof Error ? error.message : 'Please try again / Dabalataan yaali', variant: 'destructive' }); }
   201→        } else { toast({ title: 'Submission Failed / Galmeen Hin Dhufne', description: error instanceof Error ? error.message : 'Please try again / Dabalataan yaali', variant: 'destructive' }); }
   202→      } finally { setIsSubmitting(false); }
   203→    } else {
   204→      if (onOfflineSave) {
   205→        const saved = await onOfflineSave(payload);
   206→        if (saved) { toast({ title: 'Saved Offline / Offliin Kuusame', description: 'Will sync when connected / Netiirkii qunnamaa waanin dhiheenya' }); resetForm(); onSubmit(); }
   207→        else { toast({ title: 'Save Failed / Hin Kuusanne', description: 'Storage full or unavailable / Qabiyyee dhiphaachaa ykn hin jiru', variant: 'destructive' }); }
   208→      }
   209→      setIsSubmitting(false);
   210→    }
   211→  };
   212→
   213→  const reviewItems = [
   214→    { label: 'Institution / Mana Barumsaa', value: formData.institutionName },
   215→    { label: 'Institution Type / Gosa', value: formData.institutionType },
   216→    { label: 'Ownership / Abbaa Qabuu', value: formData.ownershipType },
   217→    { label: 'Sub-City / Kuttaa Maggalaa', value: formData.subCity },
   218→    { label: 'Werreda', value: formData.area },
   219→    { label: 'Address / Baka Addaa', value: formData.specificAddress },
   220→    { label: 'License No. / Lakofsaa Eyyeema', value: formData.licenseNumber || '—' },
   221→    { label: 'Students / Barataa', value: formData.studentCount },
   222→    { label: 'Teachers / Barsiisaa', value: formData.teacherCount },
   223→    { label: 'Classrooms', value: formData.classroomCount === '' ? '0' : formData.classroomCount },
   224→    { label: 'Library', value: formData.hasLibrary === 'yes' ? 'Yes / Eyee' : 'No / Lakki' },
   225→    { label: 'Laboratory', value: formData.hasLaboratory === 'yes' ? 'Yes / Eyee' : 'No / Lakki' },
   226→    { label: 'Computer Lab', value: formData.hasComputerLab === 'yes' ? 'Yes / Eyee' : 'No / Lakki' },
   227→    { label: 'Feeding Program', value: formData.hasFeedingProgram === 'yes' ? 'Yes / Eyee' : 'No / Lakki' },
   228→    { label: 'Contact / Quunnamtii', value: `${formData.contactName} — ${formData.contactPhone}` },
   229→  ];
   230→
   231→  return (
   232→    <div className="mx-auto max-w-2xl">
   233→      {/* Step Indicator */}
   234→      <div className="mb-4 sm:mb-6">
   235→        <div className="flex items-center gap-1 overflow-x-auto pb-1 sm:gap-0 sm:pb-2" style={{ scrollbarWidth: 'none', msOverflowStyle: 'none' }}>
   236→          {STEPS.map((step, idx) => {
   237→            const Icon = step.icon;
   238→            const isActive = currentStep === step.id;
   239→            const isCompleted = currentStep > step.id;
   240→            return (
   241→              <div key={step.id} className="flex flex-1 shrink-0 items-center">
   242→                <button
   243→                  type="button"
   244→                  onClick={() => isCompleted && setCurrentStep(step.id)}
   245→                  className={`flex w-full flex-col items-center gap-0.5 rounded-lg px-1 py-1.5 transition-all sm:gap-1 sm:rounded-xl sm:px-2 sm:py-2 ${
   246→                    isActive ? 'bg-indigo-600 text-white shadow-md sm:shadow-lg sm:shadow-indigo-100'
   247→                      : isCompleted ? 'bg-indigo-100 text-indigo-700 cursor-pointer'
   248→                      : 'bg-gray-100 text-gray-400'
   249→                  }`}>
   250→                  <div className={`flex h-7 w-7 items-center justify-center rounded-full sm:h-8 sm:w-8 ${isCompleted ? 'bg-indigo-600 text-white' : ''}`}>
   251→                    {isCompleted ? <Check className="h-3.5 w-3.5 sm:h-4 sm:w-4" /> : <Icon className="h-3.5 w-3.5 sm:h-4 sm:w-4" />}
   252→                  </div>
   253→                  <span className="text-[9px] leading-tight font-medium text-center truncate w-full sm:text-[10px]">{step.title}</span>
   254→                  <span className="hidden text-[8px] leading-tight text-center opacity-75 truncate w-full sm:block sm:text-[9px]">{step.titleOr}</span>
   255→                </button>
   256→                {idx < STEPS.length - 1 && (
   257→                  <div className={`mx-0.5 h-0.5 flex-1 rounded sm:mx-1 ${currentStep > step.id ? 'bg-indigo-500' : 'bg-gray-200'}`} />
   258→                )}
   259→              </div>
   260→            );
   261→          })}
   262→        </div>
   263→        <div className="mt-1.5 flex items-center justify-between text-[11px] text-muted-foreground sm:mt-2 sm:text-xs">
   264→          <span>Step {currentStep} of {STEPS.length} / Qajeelfannaa {currentStep} kiyya {STEPS.length} keessaa</span>
   265→          <button type="button" onClick={resetForm} className="flex items-center gap-1 text-red-500 hover:text-red-700">
   266→            <RotateCcw className="h-3 w-3" />
   267→            <span className="hidden sm:inline">Reset / Balleessuu</span><span className="sm:hidden">Reset</span>
   268→          </button>
   269→        </div>
   270→      </div>
   271→
   272→      <form onSubmit={handleSubmit}>
   273→        <Card className="min-h-[300px] sm:min-h-[360px]">
   274→          {/* Step 1: Institution */}
   275→          {currentStep === 1 && (
   276→            <>
   277→              <CardHeader className="pb-2 sm:pb-3">
   278→                <CardTitle className="flex items-center gap-1.5 text-base sm:gap-2 sm:text-lg">
   279→                  <GraduationCap className="h-4 w-4 shrink-0 text-indigo-600 sm:h-5 sm:w-5" />
   280→                  <span className="leading-tight">Odeeffannoo Mana Barumsaa<br /><span className="text-sm font-normal text-muted-foreground sm:text-base">Education institution information</span></span>
   281→                </CardTitle>
   282→                <CardDescription className="text-xs sm:text-sm">Basic details about the education institution</CardDescription>
   283→              </CardHeader>
   284→              <CardContent className="space-y-3 sm:space-y-4">
   285→                <div className="space-y-1.5 sm:space-y-2">
   286→                  <Label htmlFor="e-institutionName" className="text-xs sm:text-sm">Institution Name<br /><span className="text-xs font-normal text-muted-foreground">Maqaa Mana Barumsaa</span> <span className="text-red-500">*</span></Label>
   287→                  <Input id="e-institutionName" name="institutionName" placeholder="e.g., BISHOFTU PRIMARY SCHOOL" value={formData.institutionName} onChange={handleChange} required className="text-sm sm:text-base" />
   288→                </div>
   289→                <div className="space-y-1.5 sm:space-y-2">
   290→                  <Label htmlFor="e-institutionType" className="text-xs sm:text-sm">Institution Type / Gosa Mana Barumsaa <span className="text-red-500">*</span></Label>
   291→                  <Select value={formData.institutionType} onValueChange={(v) => handleSelectChange('institutionType', v)}>
   292→                    <SelectTrigger id="e-institutionType" className="text-sm sm:text-base"><SelectValue placeholder="Select / Filadhu" /></SelectTrigger>
   293→                    <SelectContent className="max-h-64">{EDUCATION_INSTITUTION_TYPES.map((t) => (<SelectItem key={t} value={t}>{t}</SelectItem>))}</SelectContent>
   294→                  </Select>
   295→                </div>
   296→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 sm:gap-4">
   297→                  <div className="space-y-1.5 sm:space-y-2">
   298→                    <Label htmlFor="e-subCity" className="text-xs sm:text-sm">Sub-City<br /><span className="text-xs font-normal text-muted-foreground">Kuttaa Maggalaa</span> <span className="text-red-500">*</span></Label>
   299→                    <Select value={formData.subCity} onValueChange={handleSubCityChange}>
   300→                      <SelectTrigger id="e-subCity" className="text-sm sm:text-base"><SelectValue placeholder="Select / Filadhu" /></SelectTrigger>
   301→                      <SelectContent>{subCities.map((sc) => (<SelectItem key={sc.name} value={sc.name}>{sc.name}</SelectItem>))}</SelectContent>
   302→                    </Select>
   303→                  </div>
   304→                  <div className="space-y-1.5 sm:space-y-2">
   305→                    <Label htmlFor="e-area" className="text-xs sm:text-sm">Werreda <span className="text-red-500">*</span></Label>
   306→                    <Select value={formData.area} onValueChange={(v) => handleSelectChange('area', v)} disabled={!selectedSubCity}>
   307→                      <SelectTrigger id="e-area" className="text-sm sm:text-base"><SelectValue placeholder={selectedSubCity ? 'Select / Filadhu' : 'Select sub-city first'} /></SelectTrigger>
   308→                      <SelectContent>{areas.map((a) => (<SelectItem key={a} value={a}>{a}</SelectItem>))}</SelectContent>
   309→                    </Select>
   310→                  </div>
   311→                </div>
   312→                <div className="space-y-1.5 sm:space-y-2">
   313→                  <Label htmlFor="e-specificAddress" className="text-xs sm:text-sm">Specific Address<br /><span className="text-xs font-normal text-muted-foreground">Baka Addaa</span> <span className="text-red-500">*</span></Label>
   314→                  <Input id="e-specificAddress" name="specificAddress" placeholder="Landmark, street, plot number" value={formData.specificAddress} onChange={handleChange} required className="text-sm sm:text-base" />
   315→                </div>
   316→              </CardContent>
   317→            </>
   318→          )}
   319→
   320→          {/* Step 2: Capacity & Services */}
   321→          {currentStep === 2 && (
   322→            <>
   323→              <CardHeader className="pb-2 sm:pb-3">
   324→                <CardTitle className="flex items-center gap-1.5 text-base sm:gap-2 sm:text-lg">
   325→                  <FileText className="h-4 w-4 shrink-0 text-indigo-600 sm:h-5 sm:w-5" />
   326→                  <span className="leading-tight">Dandeettii fi Tajaajila<br /><span className="text-sm font-normal text-muted-foreground sm:text-base">Capacity &amp; services</span></span>
   327→                </CardTitle>
   328→                <CardDescription className="text-xs sm:text-sm">License, ownership, students and available services</CardDescription>
   329→              </CardHeader>
   330→              <CardContent className="space-y-3 sm:space-y-4">
   331→                <div className="space-y-1.5 sm:space-y-2">
   332→                  <Label htmlFor="e-licenseNumber" className="text-xs sm:text-sm">License Number<br /><span className="text-xs font-normal text-muted-foreground">Lakofsaa Eyyeema</span></Label>
   333→                  <Input id="e-licenseNumber" name="licenseNumber" placeholder="e.g., ED/2015/0234" value={formData.licenseNumber} onChange={handleChange} className="text-sm sm:text-base" />
   334→                </div>
   335→                <div className="space-y-1.5 sm:space-y-2">
   336→                  <Label className="text-xs sm:text-sm">Ownership Type / Gosa Abbaa Qabuu <span className="text-red-500">*</span></Label>
   337→                  <RadioGroup value={formData.ownershipType} onValueChange={(v) => handleSelectChange('ownershipType', v)} className="grid grid-cols-1 gap-2 sm:grid-cols-2">
   338→                    {EDUCATION_OWNERSHIP_TYPES.map((t) => (
   339→                      <div key={t} className="flex items-center space-x-2 rounded-lg border p-2.5 has-data-[state=checked]:border-indigo-500 has-data-[state=checked]:bg-indigo-50">
   340→                        <RadioGroupItem value={t} id={`e-ownership-${t}`} />
   341→                        <Label htmlFor={`e-ownership-${t}`} className="cursor-pointer text-xs sm:text-sm font-normal">{t}</Label>
   342→                      </div>
   343→                    ))}
   344→                  </RadioGroup>
   345→                </div>
   346→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 sm:gap-4">
   347→                  <div className="space-y-1.5 sm:space-y-2">
   348→                    <Label htmlFor="e-studentCount" className="text-xs sm:text-sm">Student Count<br /><span className="text-xs font-normal text-muted-foreground">Lakkoofsa Barataa</span> <span className="text-red-500">*</span></Label>
   349→                    <Input id="e-studentCount" name="studentCount" type="number" min="0" placeholder="e.g., 850" value={formData.studentCount} onChange={handleChange} required className="text-sm sm:text-base" />
   350→                  </div>
   351→                  <div className="space-y-1.5 sm:space-y-2">
   352→                    <Label htmlFor="e-teacherCount" className="text-xs sm:text-sm">Teacher Count<br /><span className="text-xs font-normal text-muted-foreground">Lakkoofsa Barsiisaa</span> <span className="text-red-500">*</span></Label>
   353→                    <Input id="e-teacherCount" name="teacherCount" type="number" min="0" placeholder="e.g., 24" value={formData.teacherCount} onChange={handleChange} required className="text-sm sm:text-base" />
   354→                  </div>
   355→                </div>
   356→                <div className="space-y-1.5 sm:space-y-2">
   357→                  <Label htmlFor="e-classroomCount" className="text-xs sm:text-sm">Classroom Count <span className="text-red-500">*</span></Label>
   358→                  <Input id="e-classroomCount" name="classroomCount" type="number" min="0" placeholder="e.g., 18" value={formData.classroomCount} onChange={handleChange} required className="text-sm sm:text-base" />
   359→                </div>
   360→                {/* Available services */}
   361→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 sm:gap-4">
   362→                  {SERVICE_FIELDS.map((svc) => (
   363→                    <div key={svc.name} className="space-y-1.5 sm:space-y-2">
   364→                      <Label className="text-xs sm:text-sm">{svc.label}<br /><span className="text-xs font-normal text-muted-foreground">{svc.labelOr} — {svc.hint}</span></Label>
   365→                      <RadioGroup value={formData[svc.name as keyof typeof formData]} onValueChange={(v) => handleSelectChange(svc.name, v)} className="flex gap-4">
   366→                        <div className="flex items-center space-x-2 rounded-lg border px-3 py-2 has-data-[state=checked]:border-indigo-500 has-data-[state=checked]:bg-indigo-50">
   367→                          <RadioGroupItem value="yes" id={`e-${svc.name}-yes`} />
   368→                          <Label htmlFor={`e-${svc.name}-yes`} className="cursor-pointer text-xs sm:text-sm">Yes / Eyee</Label>
   369→                        </div>
   370→                        <div className="flex items-center space-x-2 rounded-lg border px-3 py-2 has-data-[state=checked]:border-indigo-500 has-data-[state=checked]:bg-indigo-50">
   371→                          <RadioGroupItem value="no" id={`e-${svc.name}-no`} />
