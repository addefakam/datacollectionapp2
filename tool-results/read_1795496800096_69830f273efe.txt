     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useRef } from 'react';
     4→import { Button } from '@/components/ui/button';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import {
     8→  Select,
     9→  SelectContent,
    10→  SelectItem,
    11→  SelectTrigger,
    12→  SelectValue,
    13→} from '@/components/ui/select';
    14→import {
    15→  Card,
    16→  CardContent,
    17→  CardDescription,
    18→  CardHeader,
    19→  CardTitle,
    20→} from '@/components/ui/card';
    21→import {
    22→  AlertDialog,
    23→  AlertDialogAction,
    24→  AlertDialogCancel,
    25→  AlertDialogContent,
    26→  AlertDialogDescription,
    27→  AlertDialogFooter,
    28→  AlertDialogHeader,
    29→  AlertDialogTitle,
    30→} from '@/components/ui/alert-dialog';
    31→import {
    32→  Dialog,
    33→  DialogContent,
    34→  DialogDescription,
    35→  DialogHeader,
    36→  DialogTitle,
    37→  DialogFooter,
    38→} from '@/components/ui/dialog';
    39→import { Badge } from '@/components/ui/badge';
    40→import { Skeleton } from '@/components/ui/skeleton';
    41→import { useToast } from '@/hooks/use-toast';
    42→import {
    43→  Search,
    44→  Download,
    45→  Trash2,
    46→  Eye,
    47→  Store,
    48→  ChevronLeft,
    49→  ChevronRight,
    50→  RefreshCw,
    51→  FileSpreadsheet,
    52→  Pencil,
    53→  Loader2,
    54→  Save,
    55→  Upload,
    56→  Phone,
    57→  MapPin,
    58→  FileBadge,
    59→  Coins,
    60→  Home,
    61→  GraduationCap,
    62→} from 'lucide-react';
    63→import { SMALL_ENTERPRISE_TYPES, SMALL_ENTERPRISE_OWNERSHIP_TYPES } from '@/lib/location-data';
    64→import { hasSexBreakdown } from '@/lib/sex-counts';
    65→import { useSettings } from '@/components/SettingsProvider';
    66→

    69→  enterpriseName: string;
    70→  enterpriseType: string;
    71→  ownershipType: string;
    72→  subCity: string;
    73→  area: string;
    74→  specificAddress: string;
    75→  licenseNumber: string | null;
    76→  employeeCount: number;
    77→  maleEmployees: number;
    78→  femaleEmployees: number;
    79→  startupCapital: number | null;
    80→  hasPremises: boolean;
    81→  hasTraining: boolean;
    82→  contactName: string;
    83→  contactPhone: string;
    84→  surveyorName: string | null;
    85→  surveyorId: string | null;
    86→  createdAt: string;
    87→  updatedAt: string;
    88→}
    89→
    90→interface EditForm {
    91→  enterpriseName: string;
    92→  enterpriseType: string;
    93→  ownershipType: string;
    94→  subCity: string;
    95→  area: string;
    96→  specificAddress: string;
    97→  licenseNumber: string;
    98→  employeeCount: string;
    99→  maleEmployees: string;
   100→  femaleEmployees: string;
   101→  startupCapital: string;
   102→  hasPremises: string;
   103→  hasTraining: string;
   104→  contactName: string;
   105→  contactPhone: string;
   106→}
   107→
   108→interface DataListProps {
   109→  refreshTrigger: number;
   110→}
   111→
   112→const PAGE_SIZE = 50;
   113→
   114→function fmtDate(value: string): string {
   115→  const d = new Date(value);
   116→  return isNaN(d.getTime()) ? '—' : d.toLocaleDateString('en-GB');
   117→}
   118→
   119→function fmtMoney(value: number | null): string {
   120→  if (value == null) return '—';
   121→  return value.toLocaleString('en') + ' ETB';
   122→}
   123→
   124→export default function SmallEnterpriseList({ refreshTrigger }: DataListProps) {
   125→  const { subCities, areasFor } = useSettings();
   126→  const { toast } = useToast();
   127→  const [data, setData] = useState<SmallEnterprise[]>([]);
   128→  const [loading, setLoading] = useState(true);
   129→  const [search, setSearch] = useState('');
   130→  const [filterSubCity, setFilterSubCity] = useState('');
   131→  const [filterArea, setFilterArea] = useState('');
   132→  const [filterType, setFilterType] = useState('');
   133→  const [areas, setAreas] = useState<string[]>([]);
   134→  const [page, setPage] = useState(1);
   135→  const [totalPages, setTotalPages] = useState(1);
   136→  const [total, setTotal] = useState(0);
   137→  const [deleteId, setDeleteId] = useState<string | null>(null);
   138→  const [viewItem, setViewItem] = useState<SmallEnterprise | null>(null);
   139→  const [editItem, setEditItem] = useState<SmallEnterprise | null>(null);
   140→  const [editForm, setEditForm] = useState<EditForm | null>(null);
   141→  const [editAreas, setEditAreas] = useState<string[]>([]);
   142→  const [saving, setSaving] = useState(false);
   143→  const [deleting, setDeleting] = useState(false);
   144→  const [exporting, setExporting] = useState(false);
   145→  const [importing, setImporting] = useState(false);
   146→  const [showImport, setShowImport] = useState(false);
   147→  const [importResult, setImportResult] = useState<{ success: number; failed: number; errors: string[] } | null>(null);
   148→  const fileInputRef = useRef<HTMLInputElement>(null);
   149→
   150→  const fetchData = useCallback(async () => {
   151→    setLoading(true);
   152→    try {
   153→      const params = new URLSearchParams();
   154→      if (search) params.set('search', search);
   155→      if (filterSubCity) params.set('subCity', filterSubCity);
   156→      if (filterArea) params.set('area', filterArea);
   157→      if (filterType) params.set('enterpriseType', filterType);
   158→      params.set('page', String(page));
   159→      params.set('limit', String(PAGE_SIZE));
   160→
   161→      const res = await fetch(`/api/smallenterprises?${params}`);
   162→      if (!res.ok) throw new Error('Failed to fetch');
   163→      const json = await res.json();
   164→      setData(json.data);
   165→      setTotal(json.total);
   166→      setTotalPages(json.totalPages);
   167→    } catch {
   168→      toast({
   169→        title: 'Error',
   170→        description: 'Failed to load data',
   171→        variant: 'destructive',
   172→      });
   173→    } finally {
   174→      setLoading(false);
   175→    }
   176→  }, [search, filterSubCity, filterArea, filterType, page, toast]);
   177→
   178→  useEffect(() => {
   179→    fetchData();
   180→  }, [fetchData, refreshTrigger]);
   181→
   182→  const handleSubCityFilterChange = (value: string) => {
   183→    setFilterSubCity(value);
   184→    setFilterArea('');
   185→    setPage(1);
   186→    setAreas(value ? areasFor(value) : []);
   187→  };
   188→
   189→  const handleDelete = async () => {
   190→    if (!deleteId) return;
   191→    setDeleting(true);
   192→    try {
   193→      const res = await fetch(`/api/smallenterprises/${deleteId}`, {
   194→        method: 'DELETE',
   195→      });
   196→      if (!res.ok) throw new Error('Delete failed');
   197→      toast({ title: 'Deleted', description: 'Record removed successfully.' });
   198→      fetchData();
   199→    } catch {
   200→      toast({
   201→        title: 'Delete Failed',
   202→        description: 'Could not delete the record.',
   203→        variant: 'destructive',
   204→      });
   205→    } finally {
   206→      setDeleting(false);
   207→      setDeleteId(null);
   208→    }
   209→  };
   210→
   211→  const openEdit = (item: SmallEnterprise) => {
   212→    setEditItem(item);
   213→    setEditForm({
   214→      enterpriseName: item.enterpriseName,
   215→      enterpriseType: item.enterpriseType,
   216→      ownershipType: item.ownershipType,
   217→      subCity: item.subCity,
   218→      area: item.area,
   219→      specificAddress: item.specificAddress,
   220→      licenseNumber: item.licenseNumber || '',
   221→      employeeCount: String(item.employeeCount),
   222→      maleEmployees: String(item.maleEmployees || ''),
   223→      femaleEmployees: String(item.femaleEmployees || ''),
   224→      startupCapital: item.startupCapital != null ? String(item.startupCapital) : '',
   225→      hasPremises: item.hasPremises ? 'yes' : 'no',
   226→      hasTraining: item.hasTraining ? 'yes' : 'no',
   227→      contactName: item.contactName,
   228→      contactPhone: item.contactPhone,
   229→    });
   230→    setEditAreas(areasFor(item.subCity));
   231→  };
   232→
   233→  const handleEditSubCityChange = (value: string) => {
   234→    if (!editForm) return;
   235→    setEditAreas(areasFor(value));
   236→    setEditForm({ ...editForm, subCity: value, area: '' });
   237→  };
   238→
   239→  const handleSave = async () => {
   240→    if (!editItem || !editForm) return;
   241→    setSaving(true);
   242→    try {
   243→      const res = await fetch(`/api/smallenterprises/${editItem.id}`, {
   244→        method: 'PUT',
   245→        headers: { 'Content-Type': 'application/json' },
   246→        body: JSON.stringify(editForm),
   247→      });
   248→      if (!res.ok) {
   249→        const err = await res.json().catch(() => ({}));
   250→        throw new Error(err.error || 'Update failed');
   251→      }
   252→      toast({ title: 'Saved / Nuusame', description: 'Record updated successfully' });
   253→      setEditItem(null);
   254→      setEditForm(null);
   255→      fetchData();
   256→    } catch (error) {
   257→      toast({ title: 'Error', description: error instanceof Error ? error.message : 'Failed to save', variant: 'destructive' });
   258→    } finally {
   259→      setSaving(false);
   260→    }
   261→  };
   262→
   263→  const handleExport = async (format: 'csv' | 'excel') => {
   264→    setExporting(true);
   265→    try {
   266→      const params = new URLSearchParams();
   267→      if (filterSubCity) params.set('subCity', filterSubCity);
   268→      if (filterArea) params.set('area', filterArea);
   269→      if (filterType) params.set('enterpriseType', filterType);
   270→      const res = await fetch(`/api/smallenterprises/export${format === 'excel' ? '/excel' : ''}?${params}`);
   271→      if (!res.ok) {
   272→        const err = await res.json().catch(() => ({}));
   273→        throw new Error(err.error || 'Export failed');
   274→      }
   275→      const blob = await res.blob();
   276→      const url = window.URL.createObjectURL(blob);
   277→      const a = document.createElement('a');
   278→      a.href = url;
   279→      a.download = `dhibayyuu_small_enterprise_survey_${new Date().toISOString().split('T')[0]}.${format === 'excel' ? 'xlsx' : 'csv'}`;
   280→      document.body.appendChild(a);
   281→      a.click();
   282→      document.body.removeChild(a);
   283→      window.URL.revokeObjectURL(url);
   284→      toast({ title: 'Export Successful', description: `Downloaded ${format.toUpperCase()} file` });
   285→    } catch (error) {
   286→      toast({ title: 'Export Failed', description: error instanceof Error ? error.message : 'Failed to export', variant: 'destructive' });
   287→    } finally {
   288→      setExporting(false);
   289→    }
   290→  };
   291→
   292→  const handleImport = async (e: React.ChangeEvent<HTMLInputElement>) => {
   293→    const file = e.target.files?.[0];
   294→    if (!file) return;
   295→    setImporting(true);
   296→    setImportResult(null);
   297→    try {
   298→      const formData = new FormData();
   299→      formData.append('file', file);
   300→      const res = await fetch('/api/smallenterprises/import', {
   301→        method: 'POST',
   302→        body: formData,
   303→      });
   304→      const json = await res.json();
   305→      if (!res.ok) {
   306→        throw new Error(json.error || 'Import failed');
   307→      }
   308→      setImportResult({ success: json.success, failed: json.failed, errors: json.errors || [] });
   309→      toast({
   310→        title: 'Import Complete',
   311→        description: `${json.success} records imported, ${json.failed} failed`,
   312→        variant: json.failed > 0 ? 'destructive' : 'default',
   313→      });
   314→      fetchData();
   315→    } catch (error) {
   316→      toast({ title: 'Import Failed', description: error instanceof Error ? error.message : 'Failed to import', variant: 'destructive' });
   317→    } finally {
   318→      setImporting(false);
   319→      if (fileInputRef.current) fileInputRef.current.value = '';
   320→    }
   321→  };
   322→
   323→  return (
   324→    <div className="mx-auto max-w-4xl">
   325→      <Card className="rounded-2xl border-teal-100">
   326→        <CardHeader className="pb-3">
   327→          <div className="flex flex-wrap items-center justify-between gap-2">
   328→            <CardTitle className="flex items-center gap-2 text-base sm:text-lg">
   329→              <Store className="h-5 w-5 text-amber-600" />
   330→              Small Enterprise Records
   331→              <Badge variant="secondary" className="bg-amber-100 text-amber-800">{total}</Badge>
   332→            </CardTitle>
   333→            <div className="flex gap-1.5">
   334→              <Button variant="outline" size="sm" onClick={() => fetchData()} className="h-8 px-2">
   335→                <RefreshCw className="h-3.5 w-3.5" />
   336→              </Button>
   337→              <Button variant="outline" size="sm" onClick={() => setShowImport(true)} className="h-8 px-2 text-xs">
   338→                <Upload className="h-3.5 w-3.5" />
   339→                Import
   340→              </Button>
   341→            </div>
   342→          </div>
   343→          <CardDescription className="text-xs sm:text-sm">Manage all registered small enterprises (MSE)</CardDescription>
   344→        </CardHeader>
   345→        <CardContent className="space-y-3">
   346→          {/* Filters */}
   347→          <div className="grid grid-cols-1 gap-2 sm:grid-cols-2 lg:grid-cols-4">
   348→            <div className="relative">
   349→              <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   350→              <Input
   351→                placeholder="Search name, address, contact..."
   352→                value={search}
   353→                onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   354→                className="pl-8 text-sm"
   355→              />
   356→            </div>
   357→            <Select value={filterSubCity} onValueChange={handleSubCityFilterChange}>
   358→              <SelectTrigger className="text-sm"><SelectValue placeholder="All Sub-Cities" /></SelectTrigger>
   359→              <SelectContent>{subCities.map((sc) => (<SelectItem key={sc.name} value={sc.name}>{sc.name}</SelectItem>))}</SelectContent>
   360→            </Select>
   361→            <Select value={filterArea} onValueChange={(v) => { setFilterArea(v); setPage(1); }} disabled={!filterSubCity}>
   362→              <SelectTrigger className="text-sm"><SelectValue placeholder={filterSubCity ? 'All Werredas' : 'Select sub-city first'} /></SelectTrigger>
   363→              <SelectContent>{areas.map((a) => (<SelectItem key={a} value={a}>{a}</SelectItem>))}</SelectContent>
   364→            </Select>
   365→            <Select value={filterType} onValueChange={(v) => { setFilterType(v); setPage(1); }}>
   366→              <SelectTrigger className="text-sm"><SelectValue placeholder="All Types" /></SelectTrigger>
   367→              <SelectContent className="max-h-64">{SMALL_ENTERPRISE_TYPES.map((t) => (<SelectItem key={t} value={t}>{t}</SelectItem>))}</SelectContent>
   368→            </Select>
   369→          </div>
   370→
   371→          {/* Export buttons */}
   372→          <div className="flex gap-2">
   373→            <Button variant="outline" size="sm" onClick={() => handleExport('csv')} disabled={exporting} className="flex-1 text-xs sm:text-sm">
   374→              {exporting ? <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> : <Download className="mr-1.5 h-3.5 w-3.5" />}
   375→              CSV
   376→            </Button>
   377→            <Button variant="outline" size="sm" onClick={() => handleExport('excel')} disabled={exporting} className="flex-1 text-xs sm:text-sm">
   378→              {exporting ? <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> : <FileSpreadsheet className="mr-1.5 h-3.5 w-3.5" />}
   379→              Excel
   380→            </Button>
   381→          </div>
   382→
   383→          {/* Records */}
   384→          {loading ? (
   385→            <div className="space-y-2">
   386→              {[1, 2, 3].map((i) => <Skeleton key={i} className="h-24 w-full rounded-xl" />)}
   387→            </div>
   388→          ) : data.length === 0 ? (
   389→            <div className="flex flex-col items-center gap-2 py-10 text-center">
   390→              <Store className="h-10 w-10 text-muted-foreground/40" />
   391→              <p className="text-sm font-medium text-muted-foreground">No small enterprise records found</p>
   392→              <p className="text-xs text-muted-foreground">Collect a survey, import Excel, or adjust your filters</p>
   393→            </div>
   394→          ) : (
   395→            <div className="space-y-2.5">
   396→              {data.map((item) => (
   397→                <div key={item.id} className="rounded-xl border border-teal-100 bg-white p-3 transition-shadow hover:shadow-md">
   398→                  <div className="flex flex-wrap items-start justify-between gap-2">
   399→                    <div className="min-w-0 flex-1">
   400→                      <div className="flex flex-wrap items-center gap-1.5">
   401→                        <p className="truncate text-sm font-semibold text-teal-950">{item.enterpriseName}</p>
   402→                        <Badge variant="secondary" className="bg-amber-100 text-[10px] text-amber-800">{item.enterpriseType}</Badge>
   403→                        <Badge variant="outline" className="text-[10px]">{item.ownershipType}</Badge>
   404→                        {item.licenseNumber && (
   405→                          <Badge variant="outline" className="gap-1 text-[10px] text-teal-700"><FileBadge className="h-2.5 w-2.5" />{item.licenseNumber}</Badge>
   406→                        )}
   407→                      </div>
   408→                      <div className="mt-1.5 flex flex-wrap items-center gap-x-3 gap-y-1 text-[11px] text-muted-foreground">
   409→                        <span className="flex items-center gap-1"><MapPin className="h-3 w-3" />{item.subCity} · {item.area}</span>
   410→                        <span>{item.specificAddress}</span>
   411→                        <a href={`tel:${item.contactPhone}`} className="flex items-center gap-1 font-medium text-amber-600 hover:underline">
   412→                          <Phone className="h-3 w-3" />{item.contactName}
   413→                        </a>
   414→                        <span className="text-teal-900/40">{fmtDate(item.createdAt)}</span>
   415→                      </div>
   416→                      <div className="mt-1.5 flex flex-wrap items-center gap-1.5 text-[11px]">
   417→                        <span className="inline-flex items-center gap-1 rounded-md bg-teal-50 px-1.5 py-0.5 font-medium text-teal-800">
   418→                          <Store className="h-3 w-3" />{item.employeeCount} employees
   419→                        </span>
   420→                        {hasSexBreakdown(item.maleEmployees, item.femaleEmployees) && (
   421→                          <>
   422→                            <span className="rounded-md bg-teal-50 px-1.5 py-0.5 text-teal-700">M {item.maleEmployees}</span>
   423→                            <span className="rounded-md bg-amber-50 px-1.5 py-0.5 text-amber-700">F {item.femaleEmployees}</span>
   424→                          </>
   425→                        )}
   426→                        {item.startupCapital != null && (
   427→                          <span className="inline-flex items-center gap-1 rounded-md bg-emerald-50 px-1.5 py-0.5 font-medium text-emerald-700">
   428→                            <Coins className="h-3 w-3" />{fmtMoney(item.startupCapital)}
   429→                          </span>
   430→                        )}
   431→                        {item.hasPremises && <span className="inline-flex items-center gap-1 rounded-md bg-teal-50 px-1.5 py-0.5 text-teal-700"><Home className="h-3 w-3" />Premises</span>}
   432→                        {item.hasTraining && <span className="inline-flex items-center gap-1 rounded-md bg-teal-50 px-1.5 py-0.5 text-teal-700"><GraduationCap className="h-3 w-3" />Trained</span>}
   433→                        {item.surveyorName && <span className="text-muted-foreground">by {item.surveyorName}</span>}
   434→                      </div>
   435→                    </div>
   436→                    <div className="flex shrink-0 flex-row gap-1">
   437→                      <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setViewItem(item)} title="View">
   438→                        <Eye className="h-3.5 w-3.5" />
   439→                      </Button>
   440→                      <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => openEdit(item)} title="Edit">
   441→                        <Pencil className="h-3.5 w-3.5" />
   442→                      </Button>
   443→                      <Button variant="outline" size="icon" className="h-8 w-8 border-red-200 text-red-500 hover:bg-red-50" onClick={() => setDeleteId(item.id)} title="Delete">
   444→                        <Trash2 className="h-3.5 w-3.5" />
   445→                      </Button>
   446→                    </div>
   447→                  </div>
   448→                </div>
   449→              ))}
   450→            </div>
   451→          )}
   452→
   453→          {/* Pagination */}
   454→          {totalPages > 1 && (
   455→            <div className="flex items-center justify-between pt-1">
   456→              <Button variant="outline" size="sm" onClick={() => setPage((p) => Math.max(1, p - 1))} disabled={page <= 1}>
   457→                <ChevronLeft className="h-4 w-4" /> Prev
   458→              </Button>
   459→              <span className="text-xs text-muted-foreground">Page {page} of {totalPages} · {total} records</span>
   460→              <Button variant="outline" size="sm" onClick={() => setPage((p) => Math.min(totalPages, p + 1))} disabled={page >= totalPages}>
   461→                Next <ChevronRight className="h-4 w-4" />
   462→              </Button>
   463→            </div>
   464→          )}
   465→        </CardContent>
   466→      </Card>
   467→
   468→      {/* View dialog */}
   469→      <Dialog open={!!viewItem} onOpenChange={(open) => !open && setViewItem(null)}>
   470→        <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   471→          {viewItem && (
   472→            <>
   473→              <DialogHeader>
   474→                <DialogTitle className="flex items-center gap-2">
   475→                  <Store className="h-5 w-5 text-amber-600" />
   476→                  {viewItem.enterpriseName}
   477→                </DialogTitle>
   478→                <DialogDescription>{viewItem.enterpriseType} · {viewItem.ownershipType}</DialogDescription>
   479→              </DialogHeader>
   480→              <div className="rounded-lg border bg-slate-50 text-sm">
   481→                {[
   482→                  { label: 'Sub-City / Kututtaa Maggalaa', value: viewItem.subCity },
   483→                  { label: 'Werreda', value: viewItem.area },
   484→                  { label: 'Specific Address / Baka Addaa', value: viewItem.specificAddress },
   485→                  { label: 'License No. / Lakofsaa Eyyeema', value: viewItem.licenseNumber || '—' },
   486→                  { label: 'Employees / Hojjetaa', value: String(viewItem.employeeCount) },
   487→                  { label: 'Male / Dhiira', value: String(viewItem.maleEmployees || '—') },
   488→                  { label: 'Female / Dubartii', value: String(viewItem.femaleEmployees || '—') },
   489→                  { label: 'Startup Capital', value: fmtMoney(viewItem.startupCapital) },
