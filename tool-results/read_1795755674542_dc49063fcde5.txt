     1→'use client';
     2→
     3→import { useEffect, useState, useRef } from 'react';
     4→import { Button } from '@/components/ui/button';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import {
     8→  Select,
     9→  SelectContent,
    10→  SelectItem,
    11→  SelectTrigger,
    12→  SelectValue,
    13→} from '@/components/ui/select';
    14→import {
    15→  Card,
    16→  CardContent,
    17→  CardDescription,
    18→  CardHeader,
    19→  CardTitle,
    20→} from '@/components/ui/card';
    21→import { useToast } from '@/hooks/use-toast';
    22→import {
    23→  Building2,
    24→  Loader2,
    25→  RotateCcw,
    26→  ChevronLeft,
    27→  ChevronRight,
    28→  Check,
    29→  UserCircle,
    30→  Phone,
    31→  FileText,
    32→  Eye,
    33→} from 'lucide-react';
    34→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group';
    35→import SignaturePad, { SignaturePadHandle } from '@/components/SignaturePad';
    36→import { LICENSE_TYPES, LICENSE_LEVELS } from '@/lib/location-data';
    37→import { useSettings } from '@/components/SettingsProvider';
    38→import {
    39→  sanitizeCustomAnswers,
    40→  sectorSurveyFieldEnabled,
    41→  sectorSurveyFieldRequired,
    42→  sectorSurveyFieldsDefault,
    43→  type SurveyFieldConfig,
    44→} from '@/lib/survey-fields';
    45→import {
    46→  CustomQuestionField,
    47→  customQuestionsForStep,
    48→  customReviewRows,
    49→  customStepIssue,
    50→} from '@/components/shared/custom-survey-fields';
    51→
    52→interface SurveyFormProps {
    53→  onSubmit: () => void;
    54→  surveyorName: string;
    55→  surveyorId: string;
    56→  isOnline?: boolean;
    57→  onOfflineSave?: (data: Record<string, unknown>) => Promise<boolean>;
    58→}
    59→
    60→const STEPS = [
    61→  { id: 1, title: 'Establishment', titleOr: 'Qophiin', icon: Building2 },
    62→  { id: 2, title: 'License', titleOr: 'Laisansii', icon: FileText },
    63→  { id: 3, title: 'Contact', titleOr: 'Quunnamtii', icon: Phone },
    64→  { id: 4, title: 'Review', titleOr: 'Mirkanaa', icon: Eye },
    65→];
    66→
    67→/** Uniform question label: admin-managed English label + Afaan Oromoo sub-label + required star. */
    68→function FieldLabel({ id, en, or, required }: { id: string; en: string; or?: string; required?: boolean }) {
    69→  return (
    70→    <Label htmlFor={id} className="text-xs sm:text-sm">
    71→      {en}
    72→      {or ? (<><br /><span className="text-xs font-normal text-muted-foreground">{or}</span></>) : null}
    73→      {required ? <span className="text-red-500"> *</span> : null}
    74→    </Label>
    75→  );
    76→}
    77→
    78→export default function SurveyForm({ onSubmit, surveyorName, surveyorId, isOnline = true, onOfflineSave }: SurveyFormProps) {
    79→  const { subCities, areasFor } = useSettings();
    80→  const { toast } = useToast();
    81→  const [isSubmitting, setIsSubmitting] = useState(false);
    82→  const [currentStep, setCurrentStep] = useState(1);
    83→  const [selectedSubCity, setSelectedSubCity] = useState('');
    84→  const [areas, setAreas] = useState<string[]>([]);
    85→  const [dataConfirmed, setDataConfirmed] = useState('');
    86→  const signaturePadRef = useRef<SignaturePadHandle>(null);
    87→
    88→  // Admin-managed questionnaire (Manage workspace → Survey Questions);
    89→  // registry defaults until it loads (offline keeps the defaults).
    90→  const [fields, setFields] = useState<SurveyFieldConfig[]>(() =>
    91→    sectorSurveyFieldsDefault('guesthouse')
    92→  );
    93→
    94→  useEffect(() => {
    95→    let cancelled = false;
    96→    fetch('/api/settings/survey-fields/guesthouse')
    97→      .then((r) => (r.ok ? r.json() : Promise.reject(new Error('unavailable'))))
    98→      .then((d) => {
    99→        if (!cancelled && Array.isArray(d.fields) && d.fields.length > 0) setFields(d.fields);
   100→      })
   101→      .catch(() => {
   102→        // offline — keep registry defaults
   103→      });
   104→    return () => {
   105→      cancelled = true;
   106→    };
   107→  }, []);
   108→
   109→  // Questionnaire accessors: shown / required / admin-managed bilingual label.
   110→  const shown = (key: string) => sectorSurveyFieldEnabled('guesthouse', fields, key);
   111→  const req = (key: string) => sectorSurveyFieldRequired('guesthouse', fields, key);
   112→  const label = (key: string) => {
   113→    const f = fields.find((x) => x.key === key);
   114→    return { en: f?.labelEn ?? key, or: f?.labelOr ?? '' };
   115→  };
   116→
   117→  // Admin-added custom questionnaire questions (Administration → Settings →
   118→  // Survey / Manage → Survey Questions) — answers live on the record.
   119→  const [customAnswers, setCustomAnswers] = useState<Record<string, string>>({});
   120→  const setCustom = (key: string, v: string) => setCustomAnswers((cur) => ({ ...cur, [key]: v }));
   121→
   122→  const emptyForm = {
   123→    guestHouseName: '',
   124→    organizationName: '',
   125→    subCity: '',
   126→    area: '',
   127→    specificAddress: '',
   128→    numberOfRooms: '',
   129→    licenseType: '',
   130→    licenseLevel: '',
   131→    licenseNumber: '',
   132→    contactPhone: '',
   133→    contactName: '',
   134→    contactPhone2: '',
   135→    contactName2: '',
   136→    ownerName: '',
   137→  };
   138→
   139→  const [formData, setFormData] = useState(emptyForm);
   140→
   141→  const handleSubCityChange = (value: string) => {
   142→    setSelectedSubCity(value);
   143→    setAreas(areasFor(value));
   144→    setFormData((prev) => ({ ...prev, subCity: value, area: '' }));
   145→  };
   146→
   147→  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
   148→    const { name, value } = e.target;
   149→    setFormData((prev) => ({ ...prev, [name]: value }));
   150→  };
   151→
   152→  const handleSelectChange = (name: string, value: string) => {
   153→    setFormData((prev) => ({ ...prev, [name]: value }));
   154→  };
   155→
   156→  const validateStep = (step: number): boolean => {
   157→    /** Required/format checks for the admin-added custom questions of a step. */
   158→    const checkCustom = (stepNo: number): boolean => {
   159→      const issue = customStepIssue(fields, customAnswers, stepNo);
   160→      if (!issue) return true;
   161→      toast({
   162→        title: issue.kind === 'missing' ? 'Field Missing / Qabiyyee Hin Jiru' : 'Invalid Answer / Deebii Dogoggora',
   163→        description: issue.message,
   164→        variant: 'destructive',
   165→      });
   166→      return false;
   167→    };
   168→    /** First missing required field of a step (missing = empty AND questionnaire-required). */
   169→    const missingRequired = (keys: string[]): string | null => {
   170→      for (const key of keys) {
   171→        if (req(key) && !formData[key as keyof typeof formData]) return label(key).en;
   172→      }
   173→      return null;
   174→    };
   175→    switch (step) {
   176→      case 1: {
   177→        const missing = missingRequired(['organizationName', 'subCity', 'area', 'specificAddress']);
   178→        if (missing) {
   179→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: `Required: ${missing} / Barbaachisaadha`, variant: 'destructive' });
   180→          return false;
   181→        }
   182→        return checkCustom(1);
   183→      }
   184→      case 2: {
   185→        const missing = missingRequired(['numberOfRooms', 'licenseType', 'licenseLevel', 'licenseNumber']);
   186→        if (missing) {
   187→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: `Required: ${missing} / Barbaachisaadha`, variant: 'destructive' });
   188→          return false;
   189→        }
   190→        return checkCustom(2);
   191→      }
   192→      case 3: {
   193→        const missing = missingRequired(['ownerName', 'contactName', 'contactPhone']);
   194→        if (missing) {
   195→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: `Required: ${missing} / Barbaachisaadha`, variant: 'destructive' });
   196→          return false;
   197→        }
   198→        const phoneClean = formData.contactPhone.replace(/[\s-]/g, '');
   199→        const ethPhoneRegex = /^(\+251|251|0)?(9|7)\d{8}$/;
   200→        if (!ethPhoneRegex.test(phoneClean)) {
   201→          toast({ title: 'Invalid Phone / Bilbila Dogoggora', description: 'Enter valid Ethiopian number (e.g., +251 91 234 5678)', variant: 'destructive' });
   202→          return false;
   203→        }
   204→        if (shown('contactPhone2') && formData.contactPhone2) {
   205→          const phoneClean2 = formData.contactPhone2.replace(/[\s-]/g, '');
   206→          if (!ethPhoneRegex.test(phoneClean2)) {
   207→            toast({ title: 'Invalid Phone 2 / Bilbila 2 Dogoggora', description: 'Enter valid Ethiopian number for second contact', variant: 'destructive' });
   208→            return false;
   209→          }
   210→        }
   211→        if (shown('contactPhone2') && shown('contactName2') && formData.contactPhone2 && !formData.contactName2) {
   212→          toast({ title: 'Name Required / Maqaa Barbaachisaa', description: 'Second contact person name is required when phone is provided / Bilbila 2 qofatu yoo galmeessame maqaan isaa barbaachisaa', variant: 'destructive' });
   213→          return false;
   214→        }
   215→        return checkCustom(3);
   216→      }
   217→      case 4:
   218→        if (dataConfirmed !== 'yes') {
   219→          toast({ title: 'Confirmation Required / Mirkanaa\'uu Barbaachisaa', description: 'Please confirm data is true and authenticated / Daataan dhugaa fi mirkanaa\'eef mirkanaadhu', variant: 'destructive' });
   220→          return false;
   221→        }
   222→        if (signaturePadRef.current?.isEmpty()) {
   223→          toast({ title: 'Signature Required / Mallattoo Barbaachisaa', description: 'Please provide your signature / Mallattoo keessan galmeessaa', variant: 'destructive' });
   224→          return false;
   225→        }
   226→        return true;
   227→      default:
   228→        return true;
   229→    }
   230→  };
   231→
   232→  const nextStep = () => { if (validateStep(currentStep)) setCurrentStep((s) => Math.min(s + 1, STEPS.length)); };
   233→  const prevStep = () => { setCurrentStep((s) => Math.max(s - 1, 1)); };
   234→
   235→  const resetForm = () => {
   236→    setFormData(emptyForm);
   237→    setSelectedSubCity('');
   238→    setAreas([]);
   239→    setDataConfirmed('');
   240→    setCustomAnswers({});
   241→    setCurrentStep(1);
   242→    signaturePadRef.current?.clear();
   243→  };
   244→
   245→  const handleSubmit = async (e: React.FormEvent) => {
   246→    e.preventDefault();
   247→    if (!validateStep(currentStep)) return;
   248→    const signatureData = signaturePadRef.current?.getSignature() || null;
   249→    const payload = { ...formData, serviceRating: 0, surveyorName, surveyorId, signature: signatureData, customAnswers: sanitizeCustomAnswers(fields, customAnswers) };
   250→    setIsSubmitting(true);
   251→    if (isOnline) {
   252→      try {
   253→        const response = await fetch('/api/guesthouses', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
   254→        if (!response.ok) {
   255→          const err = await response.json().catch(() => ({}));
   256→          if (response.status >= 400 && response.status < 500) {
   257→            // Server permanently rejected the record (validation) — queueing it offline would never succeed.
   258→            toast({ title: 'Submission Rejected / Galmeen Morkame', description: err.error || 'Please correct the highlighted fields and submit again', variant: 'destructive' });
   259→            return;
   260→          }
   261→          throw new Error(err.error || 'Server unavailable');
   262→        }
   263→        toast({ title: 'Survey Submitted! / Galma Qabame!', description: `${formData.organizationName} recorded successfully / galmeen safiisan argame` });
   264→        resetForm();
   265→        onSubmit();
   266→      } catch (error) {
   267→        if (onOfflineSave) {
   268→          const saved = await onOfflineSave(payload);
   269→          if (saved) { toast({ title: 'Saved Offline / Offliin Kuusame', description: 'Will sync when connected / Netiirkii qunnamaa waanin dhiheenya' }); resetForm(); onSubmit(); }
   270→          else { toast({ title: 'Submission Failed / Galmeen Hin Dhufne', description: error instanceof Error ? error.message : 'Please try again / Dabalataan yaali', variant: 'destructive' }); }
   271→        } else { toast({ title: 'Submission Failed / Galmeen Hin Dhufne', description: error instanceof Error ? error.message : 'Please try again / Dabalataan yaali', variant: 'destructive' }); }
   272→      } finally { setIsSubmitting(false); }
   273→    } else {
   274→      if (onOfflineSave) {
   275→        const saved = await onOfflineSave(payload);
   276→        if (saved) { toast({ title: 'Saved Offline / Offliin Kuusame', description: 'Will sync when connected / Netiirkii qunnamaa waanin dhiheenya' }); resetForm(); onSubmit(); }
   277→        else { toast({ title: 'Save Failed / Hin Kuusanne', description: 'Storage full or unavailable / Qabiyyee dhiphaachaa ykn hin jiru', variant: 'destructive' }); }
   278→      }
   279→      setIsSubmitting(false);
   280→    }
   281→  };
   282→
   283→  const reviewItems = (
   284→    [
   285→      shown('organizationName') && { label: 'Organization / Dhaabbataa', value: formData.organizationName },
   286→      shown('subCity') && { label: 'Sub-City / Kuttaa Maggalaa', value: formData.subCity },
   287→      shown('area') && { label: 'Werreda', value: formData.area },
   288→      shown('specificAddress') && { label: 'Address / Baka Addaa', value: formData.specificAddress },
   289→      shown('numberOfRooms') && { label: 'Rooms / Qubeettii', value: formData.numberOfRooms },
   290→      shown('licenseType') && { label: label('licenseType').en, value: formData.licenseType },
   291→      shown('licenseLevel') && { label: label('licenseLevel').en, value: formData.licenseLevel },
   292→      shown('licenseNumber') && { label: label('licenseNumber').en, value: formData.licenseNumber },
   293→      shown('ownerName') && { label: label('ownerName').en, value: formData.ownerName },
   294→      shown('contactName') && { label: 'Contact 1 / Quunnamtaa 1', value: `${formData.contactName} — ${formData.contactPhone}` },
   295→      (shown('contactName2') || shown('contactPhone2')) && { label: 'Contact 2 / Quunnamtaa 2', value: formData.contactName2 ? `${formData.contactName2} — ${formData.contactPhone2}` : formData.contactPhone2 },
   296→      ...customReviewRows(fields, customAnswers),
   297→    ].filter(Boolean) as { label: string; value: string }[]
   298→  );
   299→
   300→  return (
   301→    <div className="mx-auto max-w-2xl">
   302→      {/* Step Indicator */}
   303→      <div className="mb-4 sm:mb-6">
   304→        <div className="flex items-center gap-1 overflow-x-auto pb-1 sm:gap-0 sm:pb-2" style={{ scrollbarWidth: 'none', msOverflowStyle: 'none' }}>
   305→          {STEPS.map((step, idx) => {
   306→            const Icon = step.icon;
   307→            const isActive = currentStep === step.id;
   308→            const isCompleted = currentStep > step.id;
   309→            return (
   310→              <div key={step.id} className="flex flex-1 shrink-0 items-center">
   311→                <button
   312→                  type="button"
   313→                  onClick={() => isCompleted && setCurrentStep(step.id)}
   314→                  className={`flex w-full flex-col items-center gap-0.5 rounded-lg px-1 py-1.5 transition-all sm:gap-1 sm:rounded-xl sm:px-2 sm:py-2 ${
   315→                    isActive ? 'bg-emerald-600 text-white shadow-md sm:shadow-lg sm:shadow-emerald-200'
   316→                      : isCompleted ? 'bg-emerald-100 text-emerald-700 cursor-pointer'
   317→                      : 'bg-gray-100 text-gray-400'
   318→                  }`}>
   319→                  <div className={`flex h-7 w-7 items-center justify-center rounded-full sm:h-8 sm:w-8 ${isCompleted ? 'bg-emerald-600 text-white' : ''}`}>
   320→                    {isCompleted ? <Check className="h-3.5 w-3.5 sm:h-4 sm:w-4" /> : <Icon className="h-3.5 w-3.5 sm:h-4 sm:w-4" />}
   321→                  </div>
   322→                  <span className="text-[9px] leading-tight font-medium text-center truncate w-full sm:text-[10px]">{step.title}</span>
   323→                  <span className="hidden text-[8px] leading-tight text-center opacity-75 truncate w-full sm:block sm:text-[9px]">{step.titleOr}</span>
   324→                </button>
   325→                {idx < STEPS.length - 1 && (
   326→                  <div className={`mx-0.5 h-0.5 flex-1 rounded sm:mx-1 ${currentStep > step.id ? 'bg-emerald-500' : 'bg-gray-200'}`} />
   327→                )}
   328→              </div>
   329→            );
   330→          })}
   331→        </div>
   332→        <div className="mt-1.5 flex items-center justify-between text-[11px] text-muted-foreground sm:mt-2 sm:text-xs">
   333→          <span>Step {currentStep} of {STEPS.length} / Qajeelfannaa {currentStep} kiyya {STEPS.length} keessaa</span>
   334→          <button type="button" onClick={resetForm} className="flex items-center gap-1 text-red-500 hover:text-red-700">
   335→            <RotateCcw className="h-3 w-3" />
   336→            <span className="hidden sm:inline">Reset / Balleessuu</span><span className="sm:hidden">Reset</span>
   337→          </button>
   338→        </div>
   339→      </div>
   340→
   341→      <form onSubmit={handleSubmit}>
   342→        <Card className="min-h-[300px] sm:min-h-[360px]">
   343→          {/* Step 1: Establishment */}
   344→          {currentStep === 1 && (
   345→            <>
   346→              <CardHeader className="pb-2 sm:pb-3">
   347→                <CardTitle className="flex items-center gap-1.5 text-base sm:gap-2 sm:text-lg">
   348→                  <Building2 className="h-4 w-4 shrink-0 text-emerald-600 sm:h-5 sm:w-5" />
   349→                  <span className="leading-tight">Odeeffannoo Hundeeffama<br /><span className="text-sm font-normal text-muted-foreground sm:text-base">Establishment information</span></span>
   350→                </CardTitle>
   351→                <CardDescription className="text-xs sm:text-sm">Basic details about the establishment</CardDescription>
   352→              </CardHeader>
   353→              <CardContent className="space-y-3 sm:space-y-4">
   354→                <div className="space-y-1.5 sm:space-y-2">
   355→                  <FieldLabel id="organizationName" en={label('organizationName').en} or={label('organizationName').or} required={req('organizationName')} />
   356→                  <Input id="organizationName" name="organizationName" placeholder="e.g., GOLD MARK HOTEL" value={formData.organizationName} onChange={handleChange} required className="text-sm sm:text-base" />
   357→                </div>
   358→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 sm:gap-4">
   359→                  <div className="space-y-1.5 sm:space-y-2">
   360→                    <FieldLabel id="subCity" en={label('subCity').en} or={label('subCity').or} required={req('subCity')} />
   361→                    <Select value={formData.subCity} onValueChange={handleSubCityChange}>
   362→                      <SelectTrigger id="subCity" className="text-sm sm:text-base"><SelectValue placeholder="Select / Filadhu" /></SelectTrigger>
   363→                      <SelectContent>{subCities.map((sc) => (<SelectItem key={sc.name} value={sc.name}>{sc.name}</SelectItem>))}</SelectContent>
   364→                    </Select>
   365→                  </div>
   366→                  <div className="space-y-1.5 sm:space-y-2">
   367→                    <FieldLabel id="area" en={label('area').en} or={label('area').or} required={req('area')} />
   368→                    <Select value={formData.area} onValueChange={(v) => handleSelectChange('area', v)} disabled={!selectedSubCity}>
   369→                      <SelectTrigger id="area" className="text-sm sm:text-base"><SelectValue placeholder={selectedSubCity ? 'Select / Filadhu' : 'Select sub-city first'} /></SelectTrigger>
   370→                      <SelectContent>{areas.map((a) => (<SelectItem key={a} value={a}>{a}</SelectItem>))}</SelectContent>
   371→                    </Select>
   372→                  </div>
   373→                </div>
   374→                {shown('specificAddress') && (
   375→                <div className="space-y-1.5 sm:space-y-2">
   376→                  <FieldLabel id="specificAddress" en={label('specificAddress').en} or={label('specificAddress').or} required={req('specificAddress')} />
   377→                  <Input id="specificAddress" name="specificAddress" placeholder="Street name, landmark..." value={formData.specificAddress} onChange={handleChange} className="text-sm sm:text-base" />
   378→                </div>
   379→                )}
   380→                {customQuestionsForStep(fields, 1).map((f) => (
   381→                  <CustomQuestionField key={f.key} field={f} value={customAnswers[f.key] ?? ''} onChange={(v) => setCustom(f.key, v)} idPrefix="gh-custom" />
   382→                ))}
   383→              </CardContent>
   384→            </>
   385→          )}
   386→
   387→          {/* Step 2: License */}
   388→          {currentStep === 2 && (
   389→            <>
   390→              <CardHeader className="pb-2 sm:pb-3">
   391→                <CardTitle className="flex items-center gap-1.5 text-base sm:gap-2 sm:text-lg">
   392→                  <FileText className="h-4 w-4 shrink-0 text-blue-600 sm:h-5 sm:w-5" />
   393→                  <span className="leading-tight">Dandeettii fi Hayyama<br /><span className="text-sm font-normal text-muted-foreground sm:text-base">Capacity and license</span></span>
   394→                </CardTitle>
   395→                <CardDescription className="text-xs sm:text-sm">Room capacity and licensing information</CardDescription>
   396→              </CardHeader>
   397→              <CardContent className="space-y-3 sm:space-y-4">
   398→                <div className="space-y-1.5 sm:space-y-2">
   399→                  <FieldLabel id="numberOfRooms" en={label('numberOfRooms').en} or={label('numberOfRooms').or} required={req('numberOfRooms')} />
   400→                  <Input id="numberOfRooms" name="numberOfRooms" type="number" min="1" placeholder="e.g., 20" value={formData.numberOfRooms} onChange={handleChange} required className="text-sm sm:text-base" />
   401→                </div>
   402→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 sm:gap-4">
   403→                  {shown('licenseType') && (
   404→                  <div className="space-y-1.5 sm:space-y-2">
   405→                    <FieldLabel id="licenseType" en={label('licenseType').en} or={label('licenseType').or} required={req('licenseType')} />
   406→                    <Select value={formData.licenseType} onValueChange={(v) => handleSelectChange('licenseType', v)}>
   407→                      <SelectTrigger className="text-sm sm:text-base"><SelectValue placeholder="Select / Filadhu" /></SelectTrigger>
   408→                      <SelectContent>{LICENSE_TYPES.map((t) => (<SelectItem key={t} value={t}>{t}</SelectItem>))}</SelectContent>
   409→                    </Select>
   410→                  </div>
   411→                  )}
   412→                  {shown('licenseLevel') && (
   413→                  <div className="space-y-1.5 sm:space-y-2">
   414→                    <FieldLabel id="licenseLevel" en={label('licenseLevel').en} or={label('licenseLevel').or} required={req('licenseLevel')} />
   415→                    <Select value={formData.licenseLevel} onValueChange={(v) => handleSelectChange('licenseLevel', v)}>
   416→                      <SelectTrigger className="text-sm sm:text-base"><SelectValue placeholder="Select / Filadhu" /></SelectTrigger>
   417→                      <SelectContent>{LICENSE_LEVELS.map((l) => (<SelectItem key={l} value={l}>{l}</SelectItem>))}</SelectContent>
   418→                    </Select>
   419→                  </div>
   420→                  )}
   421→                </div>
   422→                {shown('licenseNumber') && (

   424→                  <FieldLabel id="licenseNumber" en={label('licenseNumber').en} or={label('licenseNumber').or} required={req('licenseNumber')} />
