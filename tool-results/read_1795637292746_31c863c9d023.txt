     1→'use client';
     2→
     3→import { useState } from 'react';
     4→import { Button } from '@/components/ui/button';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import {
     8→  Select,
     9→  SelectContent,
    10→  SelectItem,
    11→  SelectTrigger,
    12→  SelectValue,
    13→} from '@/components/ui/select';
    14→import {
    15→  Card,
    16→  CardContent,
    17→  CardDescription,
    18→  CardHeader,
    19→  CardTitle,
    20→} from '@/components/ui/card';
    21→import { useToast } from '@/hooks/use-toast';
    22→import {
    23→  Stethoscope,
    24→  Loader2,
    25→  RotateCcw,
    26→  ChevronLeft,
    27→  ChevronRight,
    28→  Check,
    29→  UserCircle,
    30→  Phone,
    31→  FileText,
    32→  Eye,
    33→} from 'lucide-react';
    34→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group';
    35→import { HEALTH_FACILITY_TYPES, HEALTH_OWNERSHIP_TYPES } from '@/lib/location-data';
    36→import { useSettings } from '@/components/SettingsProvider';
    37→
    38→interface HealthSurveyFormProps {
    39→  onSubmit: () => void;
    40→  surveyorName: string;
    41→  surveyorId: string;
    42→  isOnline?: boolean;
    43→  onOfflineSave?: (data: Record<string, unknown>) => Promise<boolean>;
    44→}
    45→
    46→const STEPS = [
    47→  { id: 1, title: 'Facility', titleOr: 'Manaa Yaalaa', icon: Stethoscope },
    48→  { id: 2, title: 'Capacity', titleOr: 'Dandeettii', icon: FileText },
    49→  { id: 3, title: 'Contact', titleOr: 'Quunnamtii', icon: Phone },
    50→  { id: 4, title: 'Review', titleOr: 'Mirkanaa', icon: Eye },
    51→];
    52→
    53→const SERVICE_FIELDS: { name: string; label: string; labelOr: string; hint: string }[] = [
    54→  { name: 'hasLaboratory', label: 'Laboratory service', labelOr: 'Laboratoire', hint: 'On-site lab tests' },
    55→  { name: 'hasPharmacy', label: 'Pharmacy service', labelOr: 'Faayinmaa', hint: 'Dispenses medicines' },
    56→  { name: 'hasMaternity', label: 'Maternity / delivery', labelOr: 'Dhalachaa', hint: 'Delivery & ANC services' },
    57→  { name: 'hasEmergency', label: 'Emergency (24/7)', labelOr: 'Ethiirtii 24/7', hint: 'Open day and night' },
    58→];
    59→
    60→export default function HealthSurveyForm({
    61→  onSubmit,
    62→  surveyorName,
    63→  surveyorId,
    64→  isOnline = true,
    65→  onOfflineSave,
    66→}: HealthSurveyFormProps) {
    67→  const { toast } = useToast();
    68→  const { subCities, areasFor } = useSettings();
    69→  const [isSubmitting, setIsSubmitting] = useState(false);
    70→  const [currentStep, setCurrentStep] = useState(1);
    71→  const [selectedSubCity, setSelectedSubCity] = useState('');
    72→  const [areas, setAreas] = useState<string[]>([]);
    73→  const [dataConfirmed, setDataConfirmed] = useState('');
    74→
    75→  const emptyForm = {
    76→    facilityName: '',
    77→    facilityType: '',
    78→    ownershipType: '',
    79→    subCity: '',
    80→    area: '',
    81→    specificAddress: '',
    82→    licenseNumber: '',
    83→    staffCount: '',
    84→    maleStaff: '',
    85→    femaleStaff: '',
    86→    bedCount: '',
    87→    hasLaboratory: 'no',
    88→    hasPharmacy: 'no',
    89→    hasMaternity: 'no',
    90→    hasEmergency: 'no',
    91→    contactName: '',
    92→    contactPhone: '',
    93→  };
    94→
    95→  const [formData, setFormData] = useState(emptyForm);
    96→
    97→  const handleSubCityChange = (value: string) => {
    98→    setSelectedSubCity(value);
    99→    setAreas(areasFor(value));
   100→    setFormData((prev) => ({ ...prev, subCity: value, area: '' }));
   101→  };
   102→
   103→  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
   104→    const { name, value } = e.target;
   105→    setFormData((prev) => ({ ...prev, [name]: value }));
   106→  };
   107→
   108→  const handleSelectChange = (name: string, value: string) => {
   109→    setFormData((prev) => ({ ...prev, [name]: value }));
   110→  };
   111→
   112→  /**
   113→   * Sex breakdown inputs: when both male and female are filled and the total
   114→   * is still empty, auto-fill the total with their sum.
   115→   */
   116→  const handleSexChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   117→    const { name, value } = e.target;
   118→    setFormData((prev) => {
   119→      const next = { ...prev, [name]: value } as typeof prev;
   120→      const male = name === 'maleStaff' ? value : next.maleStaff;
   121→      const female = name === 'femaleStaff' ? value : next.femaleStaff;
   122→      if (next.staffCount === '' && male !== '' && female !== '') {
   123→        const sum = Number(male) + Number(female);
   124→        if (Number.isFinite(sum) && sum >= 0) return { ...next, staffCount: String(sum) };
   125→      }
   126→      return next;
   127→    });
   128→  };
   129→
   130→  /** Shared step validation for one male/female pair against its total. */
   131→  const validateSexPair = (maleStr: string, femaleStr: string, totalStr: string, totalLabel: string): boolean => {
   132→    const male = maleStr === '' ? 0 : Number(maleStr);
   133→    const female = femaleStr === '' ? 0 : Number(femaleStr);
   134→    if (!Number.isFinite(male) || male < 0 || !Number.isFinite(female) || female < 0) {
   135→      toast({ title: 'Invalid Sex Counts / Lakkoofsa Sanyii Dogoggora', description: `Male and female ${totalLabel.toLowerCase()} must be non-negative numbers / Lakkoofsi dhiira fi dubartii lakkoofsa diidaa hin qabaatu`, variant: 'destructive' });
   136→      return false;
   137→    }
   138→    if (male + female > 0 && male + female !== Number(totalStr)) {
   139→      toast({ title: 'Sex Counts Mismatch / Lakkoofsa Sanyii Hin Wal Qabne', description: `Male + Female (${male + female}) must equal ${totalLabel} (${totalStr}) / Dhiira fi Dubartii lakkoofsa waliigalaa wajjin wal qabachuu qabu`, variant: 'destructive' });
   140→      return false;
   141→    }
   142→    return true;
   143→  };
   144→
   145→  const validateStep = (step: number): boolean => {
   146→    switch (step) {
   147→      case 1:
   148→        if (!formData.facilityName || !formData.facilityType || !formData.subCity || !formData.area || !formData.specificAddress) {
   149→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: 'All fields in this step are required / Galii hundinuu barbaachisaadha', variant: 'destructive' });
   150→          return false;
   151→        }
   152→        return true;
   153→      case 2:
   154→        if (!formData.ownershipType) {
   155→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: 'Ownership type is required / Gosa abbaa qabuu barbaachisaadha', variant: 'destructive' });
   156→          return false;
   157→        }
   158→        if (!formData.staffCount || Number(formData.staffCount) < 0) {
   159→          toast({ title: 'Invalid Staff Count / Lakkoofsa Hojjetaa Dogoggora', description: 'Enter a valid staff count / Lakkoofsa hojjetaa sirrii galchi', variant: 'destructive' });
   160→          return false;
   161→        }
   162→        if (!validateSexPair(formData.maleStaff, formData.femaleStaff, formData.staffCount, 'Staff Count')) return false;
   163→        if (formData.bedCount !== '' && Number(formData.bedCount) < 0) {
   164→          toast({ title: 'Invalid Beds / Lakkoofsa Siree Dogoggora', description: 'Bed count cannot be negative / Lakkoofsa siree diidaa hin qabaatu', variant: 'destructive' });
   165→          return false;
   166→        }
   167→        return true;
   168→      case 3:
   169→        if (!formData.contactName || !formData.contactPhone) {
   170→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: 'Contact person and phone are required / Maqaa fi bilbila quunnamtii barbaachisa', variant: 'destructive' });
   171→          return false;
   172→        }
   173→        {
   174→          const phoneClean = formData.contactPhone.replace(/[\s-]/g, '');
   175→          const ethPhoneRegex = /^(\+251|251|0)?(9|7)\d{8}$/;
   176→          if (!ethPhoneRegex.test(phoneClean)) {
   177→            toast({ title: 'Invalid Phone / Bilbila Dogoggora', description: 'Enter valid Ethiopian number (e.g., +251 91 234 5678)', variant: 'destructive' });
   178→            return false;
   179→          }
   180→        }
   181→        return true;
   182→      case 4:
   183→        if (dataConfirmed !== 'yes') {
   184→          toast({ title: 'Confirmation Required / Mirkanaa\u2019uu Barbaachisaa', description: 'Please confirm data is true and authenticated / Daataan dhugaa fi mirkanaa\u2019eef mirkanaadhu', variant: 'destructive' });
   185→          return false;
   186→        }
   187→        return true;
   188→      default:
   189→        return true;
   190→    }
   191→  };
   192→
   193→  const nextStep = () => { if (validateStep(currentStep)) setCurrentStep((s) => Math.min(s + 1, STEPS.length)); };
   194→  const prevStep = () => { setCurrentStep((s) => Math.max(s - 1, 1)); };
   195→
   196→  const resetForm = () => {
   197→    setFormData(emptyForm);
   198→    setSelectedSubCity('');
   199→    setAreas([]);
   200→    setDataConfirmed('');
   201→    setCurrentStep(1);
   202→  };
   203→
   204→  const handleSubmit = async (e: React.FormEvent) => {
   205→    e.preventDefault();
   206→    if (!validateStep(currentStep)) return;
   207→    const payload = {
   208→      ...formData,
   209→      staffCount: Number(formData.staffCount || 0),
   210→      maleStaff: formData.maleStaff === '' ? 0 : Number(formData.maleStaff),
   211→      femaleStaff: formData.femaleStaff === '' ? 0 : Number(formData.femaleStaff),
   212→      bedCount: formData.bedCount === '' ? 0 : Number(formData.bedCount),
   213→      hasLaboratory: formData.hasLaboratory === 'yes',
   214→      hasPharmacy: formData.hasPharmacy === 'yes',
   215→      hasMaternity: formData.hasMaternity === 'yes',
   216→      hasEmergency: formData.hasEmergency === 'yes',
   217→      surveyorName,
   218→      surveyorId,
   219→    };
   220→    setIsSubmitting(true);
   221→    if (isOnline) {
   222→      try {
   223→        const response = await fetch('/api/healthcenters', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
   224→        if (!response.ok) {
   225→          const err = await response.json().catch(() => ({}));
   226→          if (response.status >= 400 && response.status < 500) {
   227→            // Server permanently rejected the record (validation) — queueing it offline would never succeed.
   228→            toast({ title: 'Submission Rejected / Galmeen Morkame', description: err.error || 'Please correct the highlighted fields and submit again', variant: 'destructive' });
   229→            return;
   230→          }
   231→          throw new Error(err.error || 'Server unavailable');
   232→        }
   233→        toast({ title: 'Survey Submitted! / Galma Qabame!', description: `${formData.facilityName} recorded successfully / galmeen safiisan argame` });
   234→        resetForm();
   235→        onSubmit();
   236→      } catch (error) {
   237→        if (onOfflineSave) {
   238→          const saved = await onOfflineSave(payload);
   239→          if (saved) { toast({ title: 'Saved Offline / Offliin Kuusame', description: 'Will sync when connected / Netiirkii qunnamaa waanin dhiheenya' }); resetForm(); onSubmit(); }
   240→          else { toast({ title: 'Submission Failed / Galmeen Hin Dhufne', description: error instanceof Error ? error.message : 'Please try again / Dabalataan yaali', variant: 'destructive' }); }
   241→        } else { toast({ title: 'Submission Failed / Galmeen Hin Dhufne', description: error instanceof Error ? error.message : 'Please try again / Dabalataan yaali', variant: 'destructive' }); }
   242→      } finally { setIsSubmitting(false); }
   243→    } else {
   244→      if (onOfflineSave) {
   245→        const saved = await onOfflineSave(payload);
   246→        if (saved) { toast({ title: 'Saved Offline / Offliin Kuusame', description: 'Will sync when connected / Netiirkii qunnamaa waanin dhiheenya' }); resetForm(); onSubmit(); }
   247→        else { toast({ title: 'Save Failed / Hin Kuusanne', description: 'Storage full or unavailable / Qabiyyee dhiphaachaa ykn hin jiru', variant: 'destructive' }); }
   248→      }
   249→      setIsSubmitting(false);
   250→    }
   251→  };
   252→
   253→  const reviewItems = [
   254→    { label: 'Facility / Manaa Yaalaa', value: formData.facilityName },
   255→    { label: 'Facility Type / Gosa', value: formData.facilityType },
   256→    { label: 'Ownership / Abbaa Qabuu', value: formData.ownershipType },
   257→    { label: 'Sub-City / Kuttaa Maggalaa', value: formData.subCity },
   258→    { label: 'Werreda', value: formData.area },
   259→    { label: 'Address / Baka Addaa', value: formData.specificAddress },
   260→    { label: 'License No. / Lakofsaa Eyyeema', value: formData.licenseNumber || '—' },
   261→    { label: 'Staff / Hojjetaa', value: formData.staffCount },
   262→    { label: 'Male Staff / Dhiira', value: formData.maleStaff === '' ? '—' : formData.maleStaff },
   263→    { label: 'Female Staff / Dubartii', value: formData.femaleStaff === '' ? '—' : formData.femaleStaff },
   264→    { label: 'Beds / Siree', value: formData.bedCount === '' ? '0' : formData.bedCount },
   265→    { label: 'Laboratory', value: formData.hasLaboratory === 'yes' ? 'Yes / Eyee' : 'No / Lakki' },
   266→    { label: 'Pharmacy', value: formData.hasPharmacy === 'yes' ? 'Yes / Eyee' : 'No / Lakki' },
   267→    { label: 'Maternity / Dhalachaa', value: formData.hasMaternity === 'yes' ? 'Yes / Eyee' : 'No / Lakki' },
   268→    { label: 'Emergency (24/7)', value: formData.hasEmergency === 'yes' ? 'Yes / Eyee' : 'No / Lakki' },
   269→    { label: 'Contact / Quunnamtii', value: `${formData.contactName} — ${formData.contactPhone}` },
   270→  ];
   271→
   272→  return (
   273→    <div className="mx-auto max-w-2xl">
   274→      {/* Step Indicator */}
   275→      <div className="mb-4 sm:mb-6">
   276→        <div className="flex items-center gap-1 overflow-x-auto pb-1 sm:gap-0 sm:pb-2" style={{ scrollbarWidth: 'none', msOverflowStyle: 'none' }}>
   277→          {STEPS.map((step, idx) => {
   278→            const Icon = step.icon;
   279→            const isActive = currentStep === step.id;
   280→            const isCompleted = currentStep > step.id;
   281→            return (
   282→              <div key={step.id} className="flex flex-1 shrink-0 items-center">
   283→                <button
   284→                  type="button"
   285→                  onClick={() => isCompleted && setCurrentStep(step.id)}
   286→                  className={`flex w-full flex-col items-center gap-0.5 rounded-lg px-1 py-1.5 transition-all sm:gap-1 sm:rounded-xl sm:px-2 sm:py-2 ${
   287→                    isActive ? 'bg-sky-600 text-white shadow-md sm:shadow-lg sm:shadow-sky-100'
   288→                      : isCompleted ? 'bg-sky-100 text-sky-700 cursor-pointer'
   289→                      : 'bg-gray-100 text-gray-400'
   290→                  }`}>
   291→                  <div className={`flex h-7 w-7 items-center justify-center rounded-full sm:h-8 sm:w-8 ${isCompleted ? 'bg-sky-600 text-white' : ''}`}>
   292→                    {isCompleted ? <Check className="h-3.5 w-3.5 sm:h-4 sm:w-4" /> : <Icon className="h-3.5 w-3.5 sm:h-4 sm:w-4" />}
   293→                  </div>
   294→                  <span className="text-[9px] leading-tight font-medium text-center truncate w-full sm:text-[10px]">{step.title}</span>
   295→                  <span className="hidden text-[8px] leading-tight text-center opacity-75 truncate w-full sm:block sm:text-[9px]">{step.titleOr}</span>
   296→                </button>
   297→                {idx < STEPS.length - 1 && (
   298→                  <div className={`mx-0.5 h-0.5 flex-1 rounded sm:mx-1 ${currentStep > step.id ? 'bg-sky-500' : 'bg-gray-200'}`} />
   299→                )}
   300→              </div>
   301→            );
   302→          })}
   303→        </div>
   304→        <div className="mt-1.5 flex items-center justify-between text-[11px] text-muted-foreground sm:mt-2 sm:text-xs">
   305→          <span>Step {currentStep} of {STEPS.length} / Qajeelfannaa {currentStep} kiyya {STEPS.length} keessaa</span>
   306→          <button type="button" onClick={resetForm} className="flex items-center gap-1 text-red-500 hover:text-red-700">
   307→            <RotateCcw className="h-3 w-3" />
   308→            <span className="hidden sm:inline">Reset / Balleessuu</span><span className="sm:hidden">Reset</span>
   309→          </button>
   310→        </div>
   311→      </div>
   312→
   313→      <form onSubmit={handleSubmit}>
   314→        <Card className="min-h-[300px] sm:min-h-[360px]">
   315→          {/* Step 1: Facility */}
   316→          {currentStep === 1 && (
   317→            <>
   318→              <CardHeader className="pb-2 sm:pb-3">
   319→                <CardTitle className="flex items-center gap-1.5 text-base sm:gap-2 sm:text-lg">
   320→                  <Stethoscope className="h-4 w-4 shrink-0 text-sky-600 sm:h-5 sm:w-5" />
   321→                  <span className="leading-tight">Odeeffannoo Manaa Yaalaa<br /><span className="text-sm font-normal text-muted-foreground sm:text-base">Health facility information</span></span>
   322→                </CardTitle>
   323→                <CardDescription className="text-xs sm:text-sm">Basic details about the health facility</CardDescription>
   324→              </CardHeader>
   325→              <CardContent className="space-y-3 sm:space-y-4">
   326→                <div className="space-y-1.5 sm:space-y-2">
   327→                  <Label htmlFor="facilityName" className="text-xs sm:text-sm">Facility Name<br /><span className="text-xs font-normal text-muted-foreground">Maqaa Manaa Yaalaa</span> <span className="text-red-500">*</span></Label>
   328→                  <Input id="facilityName" name="facilityName" placeholder="e.g., BISHOFTU GENERAL HOSPITAL" value={formData.facilityName} onChange={handleChange} required className="text-sm sm:text-base" />
   329→                </div>
   330→                <div className="space-y-1.5 sm:space-y-2">
   331→                  <Label htmlFor="facilityType" className="text-xs sm:text-sm">Facility Type / Gosa Manaa Yaalaa <span className="text-red-500">*</span></Label>
   332→                  <Select value={formData.facilityType} onValueChange={(v) => handleSelectChange('facilityType', v)}>
   333→                    <SelectTrigger id="facilityType" className="text-sm sm:text-base"><SelectValue placeholder="Select / Filadhu" /></SelectTrigger>
   334→                    <SelectContent className="max-h-64">{HEALTH_FACILITY_TYPES.map((t) => (<SelectItem key={t} value={t}>{t}</SelectItem>))}</SelectContent>
   335→                  </Select>
   336→                </div>
   337→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 sm:gap-4">
   338→                  <div className="space-y-1.5 sm:space-y-2">
   339→                    <Label htmlFor="subCity" className="text-xs sm:text-sm">Sub-City<br /><span className="text-xs font-normal text-muted-foreground">Kuttaa Maggalaa</span> <span className="text-red-500">*</span></Label>
   340→                    <Select value={formData.subCity} onValueChange={handleSubCityChange}>
   341→                      <SelectTrigger id="subCity" className="text-sm sm:text-base"><SelectValue placeholder="Select / Filadhu" /></SelectTrigger>
   342→                      <SelectContent>{subCities.map((sc) => (<SelectItem key={sc.name} value={sc.name}>{sc.name}</SelectItem>))}</SelectContent>
   343→                    </Select>
   344→                  </div>
   345→                  <div className="space-y-1.5 sm:space-y-2">
   346→                    <Label htmlFor="area" className="text-xs sm:text-sm">Werreda <span className="text-red-500">*</span></Label>
   347→                    <Select value={formData.area} onValueChange={(v) => handleSelectChange('area', v)} disabled={!selectedSubCity}>
   348→                      <SelectTrigger id="area" className="text-sm sm:text-base"><SelectValue placeholder={selectedSubCity ? 'Select / Filadhu' : 'Select sub-city first'} /></SelectTrigger>
   349→                      <SelectContent>{areas.map((a) => (<SelectItem key={a} value={a}>{a}</SelectItem>))}</SelectContent>
   350→                    </Select>
   351→                  </div>
   352→                </div>
   353→                <div className="space-y-1.5 sm:space-y-2">
   354→                  <Label htmlFor="specificAddress" className="text-xs sm:text-sm">Specific Address<br /><span className="text-xs font-normal text-muted-foreground">Baka Addaa</span> <span className="text-red-500">*</span></Label>
   355→                  <Input id="specificAddress" name="specificAddress" placeholder="Landmark, street, plot number" value={formData.specificAddress} onChange={handleChange} required className="text-sm sm:text-base" />
   356→                </div>
   357→              </CardContent>
   358→            </>
   359→          )}
   360→
   361→          {/* Step 2: Capacity & Services */}
   362→          {currentStep === 2 && (
   363→            <>
   364→              <CardHeader className="pb-2 sm:pb-3">
   365→                <CardTitle className="flex items-center gap-1.5 text-base sm:gap-2 sm:text-lg">
   366→                  <FileText className="h-4 w-4 shrink-0 text-sky-600 sm:h-5 sm:w-5" />
   367→                  <span className="leading-tight">Dandeettii fi Tajaajila<br /><span className="text-sm font-normal text-muted-foreground sm:text-base">Capacity &amp; services</span></span>
   368→                </CardTitle>
   369→                <CardDescription className="text-xs sm:text-sm">License, ownership, staff and available services</CardDescription>
   370→              </CardHeader>
   371→              <CardContent className="space-y-3 sm:space-y-4">
   372→                <div className="space-y-1.5 sm:space-y-2">
   373→                  <Label htmlFor="licenseNumber" className="text-xs sm:text-sm">License Number<br /><span className="text-xs font-normal text-muted-foreground">Lakofsaa Eyyeema</span></Label>
   374→                  <Input id="licenseNumber" name="licenseNumber" placeholder="e.g., HC/2015/0234" value={formData.licenseNumber} onChange={handleChange} className="text-sm sm:text-base" />
   375→                </div>
   376→                <div className="space-y-1.5 sm:space-y-2">
   377→                  <Label className="text-xs sm:text-sm">Ownership Type / Gosa Abbaa Qabuu <span className="text-red-500">*</span></Label>
   378→                  <RadioGroup value={formData.ownershipType} onValueChange={(v) => handleSelectChange('ownershipType', v)} className="grid grid-cols-1 gap-2 sm:grid-cols-2">
   379→                    {HEALTH_OWNERSHIP_TYPES.map((t) => (
   380→                      <div key={t} className="flex items-center space-x-2 rounded-lg border p-2.5 has-data-[state=checked]:border-sky-500 has-data-[state=checked]:bg-sky-50">
   381→                        <RadioGroupItem value={t} id={`h-ownership-${t}`} />
   382→                        <Label htmlFor={`h-ownership-${t}`} className="cursor-pointer text-xs sm:text-sm font-normal">{t}</Label>
   383→                      </div>
   384→                    ))}
   385→                  </RadioGroup>
   386→                </div>
   387→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 sm:gap-4">
