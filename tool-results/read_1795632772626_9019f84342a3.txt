   359→
   360→  // Recomputed per render as plain code (cheap at dashboard scale; a hook here would sit below the loading early-returns).
   361→  const report = (() => {
   362→    if (!data) return null;
   363→    const scopeLabel = data.scope.level === 'city' ? 'City-wide' : `Sub-city: ${data.scope.subCity}`;
   364→    const subtitle = `${data.period.label} · ${scopeLabel}`;
   365→    const statusKeys: SmeStatus[] = ['REGISTERED', 'FULLY_LEGALIZED', 'AWAITING_OPERATION', 'OPERATIONAL'];
   366→    const col = (k: ReportColumnKey) => reportCols.includes(k);
   367→    const sumAt = (rows: (string | number)[][], i: number) => rows.reduce((s, row) => s + (typeof row[i] === 'number' ? (row[i] as number) : 0), 0);
   368→
   369→    if (reportGroup === 'status') {

   389→    if (col('pending')) cols.push('Pending');

   391→    if (col('members')) cols.push('Members', 'Female members');

   401→        if (col('achievement')) row.push(r.target > 0 ? `${r.achievementPct}%` : '—');

   404→        if (col('members')) row.push(r.members, r.femaleMembers);
   405→        return row;
   406→      });
   407→      let ci = 1;
   408→      if (col('registered')) { totals.push(sumAt(rows, ci)); ci += 1; }
   409→      if (col('target')) { totals.push(sumAt(rows, ci)); ci += 1; }
   410→      if (col('achievement')) { totals.push(`${data.kpis.achievementPct}%`); ci += 1; }
   411→      if (col('pending')) { totals.push(sumAt(rows, ci)); ci += 1; }
   412→      if (col('stages')) { totals.push(sumAt(rows, ci), sumAt(rows, ci + 1), sumAt(rows, ci + 2), sumAt(rows, ci + 3), sumAt(rows, ci + 4)); ci += 5; }
   413→      if (col('members')) { totals.push(sumAt(rows, ci), sumAt(rows, ci + 1)); }
   414→    } else {

   416→        const row: (string | number)[] = [t.type];

   418→        if (col('stages')) row.push(t.standing.REGISTERED, t.standing.FULLY_LEGALIZED, t.standing.AWAITING_OPERATION, t.standing.OPERATIONAL, t.standingTotal);
   419→        if (col('members')) row.push(t.members, t.femaleMembers);
   420→        return row;
   421→      });
   422→      let ci = 1;
   423→      if (col('registered')) { totals.push(sumAt(rows, ci)); ci += 1; }
   424→      if (col('stages')) { totals.push(sumAt(rows, ci), sumAt(rows, ci + 1), sumAt(rows, ci + 2), sumAt(rows, ci + 3), sumAt(rows, ci + 4)); ci += 5; }
   425→      if (col('members')) { totals.push(sumAt(rows, ci), sumAt(rows, ci + 1)); }
   426→    }
   427→
   428→    return { title: REPORT_TITLES[reportGroup], subtitle, cols, rows, totals };

   431→  const exportReportCsv = () => {
   432→    if (!report || report.rows.length === 0) return;
   433→    const cell = (v: string | number) => {
   434→      const s = String(v ?? '');
   435→      return /[",\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
   436→    };
   437→    const lines: (string | number)[][] = [report.cols, ...report.rows];
   438→    if (report.totals) lines.push(report.totals);
   439→    const csv = lines.map((row) => row.map(cell).join(',')).join('\r\n');
   440→    const blob = new Blob([`\uFEFF${csv}`], { type: 'text/csv;charset=utf-8' });
   441→    const url = URL.createObjectURL(blob);
   442→    const a = document.createElement('a');
   443→    a.href = url;
   444→    a.download = `sme-custom-report-${new Date().toISOString().slice(0, 10)}.csv`;
   445→    document.body.appendChild(a);
   446→    a.click();
   447→    a.remove();
   448→    URL.revokeObjectURL(url);
   449→    toast({ title: 'Report Exported', description: 'CSV downloaded — it opens cleanly in Excel.' });
   450→  };

   452→  const printReport = () => {
   453→    if (!report) return;

   456→      .map((c, i) => `<th style="padding:6px 10px;border-bottom:2px solid #0d9488;text-align:${i === 0 ? 'left' : 'right'};font-size:11px;text-transform:uppercase;color:#134e4a">${esc(c)}</th>`)
   457→      .join('');
   458→    const trs = report.rows
   459→      .map((r) => `<tr>${r.map((c, i) => `<td style="padding:6px 10px;border-bottom:1px solid #e2e8f0;text-align:${i === 0 ? 'left' : 'right'};font-weight:${i === 0 ? 600 : 400}">${esc(c)}</td>`).join('')}</tr>`)
   460→      .join('');
   461→    const totalsHtml = report.totals
   462→      ? `<tr style="background:#f0fdfa;font-weight:700">${report.totals.map((c, i) => `<td style="padding:6px 10px;border-top:2px solid #0d9488;text-align:${i === 0 ? 'left' : 'right'}">${esc(c)}</td>`).join('')}</tr>`
   463→      : '';
   464→    const html = [
   465→      '<!doctype html><html><head><meta charset="utf-8" />',
   466→      `<title>${esc(report.title)} — SME Performance</title>`,
   467→      '<style>body{font-family:Arial,Helvetica,sans-serif;color:#0f172a;margin:24px}h1{font-size:18px;margin:0 0 4px}p{margin:0 0 16px;font-size:12px;color:#64748b}table{border-collapse:collapse;width:100%;font-size:12px}</style>',
   468→      '</head><body>',
   469→      `<h1>${esc(report.title)} — SME Performance</h1>`,
   470→      `<p>${esc(report.subtitle)} · generated ${new Date().toLocaleString('en-GB')}</p>`,
   471→      `<table><thead><tr>${th}</tr></thead><tbody>${trs}${totalsHtml}</tbody></table>`,
   472→      '</body></html>',
   473→    ].join('');
   474→    const w = window.open('', '_blank');
   475→    if (!w) {
   476→      toast({ title: 'Popup Blocked', description: 'Allow pop-ups for this site to print the report.', variant: 'destructive' });
   477→      return;
