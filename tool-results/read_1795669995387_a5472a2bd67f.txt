     1→'use client';
     2→
     3→import { useEffect, useState, useRef } from 'react';
     4→import { Button } from '@/components/ui/button';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import {
     8→  Select,
     9→  SelectContent,
    10→  SelectItem,
    11→  SelectTrigger,
    12→  SelectValue,
    13→} from '@/components/ui/select';
    14→import {
    15→  Card,
    16→  CardContent,
    17→  CardDescription,
    18→  CardHeader,
    19→  CardTitle,
    20→} from '@/components/ui/card';
    21→import { useToast } from '@/hooks/use-toast';
    22→import {
    23→  Building2,
    24→  Loader2,
    25→  RotateCcw,
    26→  ChevronLeft,
    27→  ChevronRight,
    28→  Check,
    29→  UserCircle,
    30→  Phone,
    31→  FileText,
    32→  Eye,
    33→} from 'lucide-react';
    34→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group';
    35→import SignaturePad, { SignaturePadHandle } from '@/components/SignaturePad';
    36→import { LICENSE_TYPES, LICENSE_LEVELS } from '@/lib/location-data';
    37→import { useSettings } from '@/components/SettingsProvider';
    38→import {
    39→  sectorSurveyFieldEnabled,
    40→  sectorSurveyFieldRequired,
    41→  sectorSurveyFieldsDefault,
    42→  type SurveyFieldConfig,
    43→} from '@/lib/survey-fields';
    44→
    45→interface SurveyFormProps {
    46→  onSubmit: () => void;
    47→  surveyorName: string;
    48→  surveyorId: string;
    49→  isOnline?: boolean;
    50→  onOfflineSave?: (data: Record<string, unknown>) => Promise<boolean>;
    51→}
    52→
    53→const STEPS = [
    54→  { id: 1, title: 'Establishment', titleOr: 'Qophiin', icon: Building2 },
    55→  { id: 2, title: 'License', titleOr: 'Laisansii', icon: FileText },
    56→  { id: 3, title: 'Contact', titleOr: 'Quunnamtii', icon: Phone },
    57→  { id: 4, title: 'Review', titleOr: 'Mirkanaa', icon: Eye },
    58→];
    59→
    60→/** Uniform question label: admin-managed English label + Afaan Oromoo sub-label + required star. */
    61→function FieldLabel({ id, en, or, required }: { id: string; en: string; or?: string; required?: boolean }) {
    62→  return (
    63→    <Label htmlFor={id} className="text-xs sm:text-sm">
    64→      {en}
    65→      {or ? (<><br /><span className="text-xs font-normal text-muted-foreground">{or}</span></>) : null}
    66→      {required ? <span className="text-red-500"> *</span> : null}
    67→    </Label>
    68→  );
    69→}
    70→
    71→export default function SurveyForm({ onSubmit, surveyorName, surveyorId, isOnline = true, onOfflineSave }: SurveyFormProps) {
    72→  const { subCities, areasFor } = useSettings();
    73→  const { toast } = useToast();
    74→  const [isSubmitting, setIsSubmitting] = useState(false);
    75→  const [currentStep, setCurrentStep] = useState(1);
    76→  const [selectedSubCity, setSelectedSubCity] = useState('');
    77→  const [areas, setAreas] = useState<string[]>([]);
    78→  const [dataConfirmed, setDataConfirmed] = useState('');
    79→  const signaturePadRef = useRef<SignaturePadHandle>(null);
    80→
    81→  // Admin-managed questionnaire (Manage workspace → Survey Questions);
    82→  // registry defaults until it loads (offline keeps the defaults).
    83→  const [fields, setFields] = useState<SurveyFieldConfig[]>(() =>
    84→    sectorSurveyFieldsDefault('guesthouse')
    85→  );
    86→
    87→  useEffect(() => {
    88→    let cancelled = false;
    89→    fetch('/api/settings/survey-fields/guesthouse')
    90→      .then((r) => (r.ok ? r.json() : Promise.reject(new Error('unavailable'))))
    91→      .then((d) => {
    92→        if (!cancelled && Array.isArray(d.fields) && d.fields.length > 0) setFields(d.fields);
    93→      })
    94→      .catch(() => {
    95→        // offline — keep registry defaults
    96→      });
    97→    return () => {
    98→      cancelled = true;
    99→    };
   100→  }, []);
   101→
   102→  // Questionnaire accessors: shown / required / admin-managed bilingual label.
   103→  const shown = (key: string) => sectorSurveyFieldEnabled('guesthouse', fields, key);
   104→  const req = (key: string) => sectorSurveyFieldRequired('guesthouse', fields, key);
   105→  const label = (key: string) => {
   106→    const f = fields.find((x) => x.key === key);
   107→    return { en: f?.labelEn ?? key, or: f?.labelOr ?? '' };
   108→  };
   109→
   110→  const emptyForm = {
   111→    guestHouseName: '',
   112→    organizationName: '',
   113→    subCity: '',
   114→    area: '',
   115→    specificAddress: '',
   116→    numberOfRooms: '',
   117→    licenseType: '',
   118→    licenseLevel: '',
   119→    licenseNumber: '',
   120→    contactPhone: '',
   121→    contactName: '',
   122→    contactPhone2: '',
   123→    contactName2: '',
   124→    ownerName: '',
   125→  };
   126→
   127→  const [formData, setFormData] = useState(emptyForm);
   128→
   129→  const handleSubCityChange = (value: string) => {
   130→    setSelectedSubCity(value);
   131→    setAreas(areasFor(value));
   132→    setFormData((prev) => ({ ...prev, subCity: value, area: '' }));
   133→  };
   134→
   135→  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
   136→    const { name, value } = e.target;
   137→    setFormData((prev) => ({ ...prev, [name]: value }));
   138→  };
   139→
   140→  const handleSelectChange = (name: string, value: string) => {
   141→    setFormData((prev) => ({ ...prev, [name]: value }));
   142→  };
   143→
   144→  const validateStep = (step: number): boolean => {
   145→    /** First missing required field of a step (missing = empty AND questionnaire-required). */
   146→    const missingRequired = (keys: string[]): string | null => {
   147→      for (const key of keys) {
   148→        if (req(key) && !formData[key as keyof typeof formData]) return label(key).en;
   149→      }
   150→      return null;
   151→    };
   152→    switch (step) {
   153→      case 1: {
   154→        const missing = missingRequired(['organizationName', 'subCity', 'area', 'specificAddress']);
   155→        if (missing) {
   156→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: `Required: ${missing} / Barbaachisaadha`, variant: 'destructive' });
   157→          return false;
   158→        }
   159→        return true;
   160→      }
   161→      case 2: {
   162→        const missing = missingRequired(['numberOfRooms', 'licenseType', 'licenseLevel', 'licenseNumber']);
   163→        if (missing) {
   164→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: `Required: ${missing} / Barbaachisaadha`, variant: 'destructive' });
   165→          return false;
   166→        }
   167→        return true;
   168→      }
   169→      case 3: {
   170→        const missing = missingRequired(['ownerName', 'contactName', 'contactPhone']);
   171→        if (missing) {
   172→          toast({ title: 'Field Missing / Qabiyyee Hin Jiru', description: `Required: ${missing} / Barbaachisaadha`, variant: 'destructive' });
   173→          return false;
   174→        }
   175→        const phoneClean = formData.contactPhone.replace(/[\s-]/g, '');
   176→        const ethPhoneRegex = /^(\+251|251|0)?(9|7)\d{8}$/;
   177→        if (!ethPhoneRegex.test(phoneClean)) {
   178→          toast({ title: 'Invalid Phone / Bilbila Dogoggora', description: 'Enter valid Ethiopian number (e.g., +251 91 234 5678)', variant: 'destructive' });
   179→          return false;
   180→        }
   181→        if (shown('contactPhone2') && formData.contactPhone2) {
   182→          const phoneClean2 = formData.contactPhone2.replace(/[\s-]/g, '');
   183→          if (!ethPhoneRegex.test(phoneClean2)) {
   184→            toast({ title: 'Invalid Phone 2 / Bilbila 2 Dogoggora', description: 'Enter valid Ethiopian number for second contact', variant: 'destructive' });
   185→            return false;
   186→          }
   187→        }
   188→        if (shown('contactPhone2') && shown('contactName2') && formData.contactPhone2 && !formData.contactName2) {
   189→          toast({ title: 'Name Required / Maqaa Barbaachisaa', description: 'Second contact person name is required when phone is provided / Bilbila 2 qofatu yoo galmeessame maqaan isaa barbaachisaa', variant: 'destructive' });
   190→          return false;
   191→        }
   192→        return true;
   193→      }
   194→      case 4:
   195→        if (dataConfirmed !== 'yes') {
   196→          toast({ title: 'Confirmation Required / Mirkanaa\'uu Barbaachisaa', description: 'Please confirm data is true and authenticated / Daataan dhugaa fi mirkanaa\'eef mirkanaadhu', variant: 'destructive' });
   197→          return false;
   198→        }
   199→        if (signaturePadRef.current?.isEmpty()) {
   200→          toast({ title: 'Signature Required / Mallattoo Barbaachisaa', description: 'Please provide your signature / Mallattoo keessan galmeessaa', variant: 'destructive' });
   201→          return false;
   202→        }
   203→        return true;
   204→      default:
   205→        return true;
   206→    }
   207→  };
   208→
   209→  const nextStep = () => { if (validateStep(currentStep)) setCurrentStep((s) => Math.min(s + 1, STEPS.length)); };
   210→  const prevStep = () => { setCurrentStep((s) => Math.max(s - 1, 1)); };
   211→
   212→  const resetForm = () => {
   213→    setFormData(emptyForm);
   214→    setSelectedSubCity('');
   215→    setAreas([]);
   216→    setDataConfirmed('');
   217→    setCurrentStep(1);
   218→    signaturePadRef.current?.clear();
   219→  };
   220→
   221→  const handleSubmit = async (e: React.FormEvent) => {
   222→    e.preventDefault();
   223→    if (!validateStep(currentStep)) return;
   224→    const signatureData = signaturePadRef.current?.getSignature() || null;
   225→    const payload = { ...formData, serviceRating: 0, surveyorName, surveyorId, signature: signatureData };
   226→    setIsSubmitting(true);
   227→    if (isOnline) {
   228→      try {
   229→        const response = await fetch('/api/guesthouses', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
   230→        if (!response.ok) {
   231→          const err = await response.json().catch(() => ({}));
   232→          if (response.status >= 400 && response.status < 500) {
   233→            // Server permanently rejected the record (validation) — queueing it offline would never succeed.
   234→            toast({ title: 'Submission Rejected / Galmeen Morkame', description: err.error || 'Please correct the highlighted fields and submit again', variant: 'destructive' });
   235→            return;
   236→          }
   237→          throw new Error(err.error || 'Server unavailable');
   238→        }
   239→        toast({ title: 'Survey Submitted! / Galma Qabame!', description: `${formData.organizationName} recorded successfully / galmeen safiisan argame` });
   240→        resetForm();
   241→        onSubmit();
   242→      } catch (error) {
   243→        if (onOfflineSave) {
   244→          const saved = await onOfflineSave(payload);
   245→          if (saved) { toast({ title: 'Saved Offline / Offliin Kuusame', description: 'Will sync when connected / Netiirkii qunnamaa waanin dhiheenya' }); resetForm(); onSubmit(); }
   246→          else { toast({ title: 'Submission Failed / Galmeen Hin Dhufne', description: error instanceof Error ? error.message : 'Please try again / Dabalataan yaali', variant: 'destructive' }); }
   247→        } else { toast({ title: 'Submission Failed / Galmeen Hin Dhufne', description: error instanceof Error ? error.message : 'Please try again / Dabalataan yaali', variant: 'destructive' }); }
   248→      } finally { setIsSubmitting(false); }
   249→    } else {
   250→      if (onOfflineSave) {
   251→        const saved = await onOfflineSave(payload);
   252→        if (saved) { toast({ title: 'Saved Offline / Offliin Kuusame', description: 'Will sync when connected / Netiirkii qunnamaa waanin dhiheenya' }); resetForm(); onSubmit(); }
   253→        else { toast({ title: 'Save Failed / Hin Kuusanne', description: 'Storage full or unavailable / Qabiyyee dhiphaachaa ykn hin jiru', variant: 'destructive' }); }
   254→      }
   255→      setIsSubmitting(false);
   256→    }
   257→  };
   258→
   259→  const reviewItems = (
   260→    [
   261→      shown('organizationName') && { label: 'Organization / Dhaabbataa', value: formData.organizationName },
   262→      shown('subCity') && { label: 'Sub-City / Kuttaa Maggalaa', value: formData.subCity },
   263→      shown('area') && { label: 'Werreda', value: formData.area },
   264→      shown('specificAddress') && { label: 'Address / Baka Addaa', value: formData.specificAddress },
   265→      shown('numberOfRooms') && { label: 'Rooms / Qubeettii', value: formData.numberOfRooms },
   266→      shown('licenseType') && { label: label('licenseType').en, value: formData.licenseType },
   267→      shown('licenseLevel') && { label: label('licenseLevel').en, value: formData.licenseLevel },
   268→      shown('licenseNumber') && { label: label('licenseNumber').en, value: formData.licenseNumber },
   269→      shown('ownerName') && { label: label('ownerName').en, value: formData.ownerName },
   270→      shown('contactName') && { label: 'Contact 1 / Quunnamtaa 1', value: `${formData.contactName} — ${formData.contactPhone}` },
   271→      (shown('contactName2') || shown('contactPhone2')) && { label: 'Contact 2 / Quunnamtaa 2', value: formData.contactName2 ? `${formData.contactName2} — ${formData.contactPhone2}` : formData.contactPhone2 },
   272→    ].filter(Boolean) as { label: string; value: string }[]
   273→  );
   274→
   275→  return (
   276→    <div className="mx-auto max-w-2xl">
   277→      {/* Step Indicator */}
   278→      <div className="mb-4 sm:mb-6">
   279→        <div className="flex items-center gap-1 overflow-x-auto pb-1 sm:gap-0 sm:pb-2" style={{ scrollbarWidth: 'none', msOverflowStyle: 'none' }}>
   280→          {STEPS.map((step, idx) => {
   281→            const Icon = step.icon;
   282→            const isActive = currentStep === step.id;
   283→            const isCompleted = currentStep > step.id;
   284→            return (
   285→              <div key={step.id} className="flex flex-1 shrink-0 items-center">
   286→                <button
   287→                  type="button"
   288→                  onClick={() => isCompleted && setCurrentStep(step.id)}
   289→                  className={`flex w-full flex-col items-center gap-0.5 rounded-lg px-1 py-1.5 transition-all sm:gap-1 sm:rounded-xl sm:px-2 sm:py-2 ${
   290→                    isActive ? 'bg-emerald-600 text-white shadow-md sm:shadow-lg sm:shadow-emerald-200'
   291→                      : isCompleted ? 'bg-emerald-100 text-emerald-700 cursor-pointer'
   292→                      : 'bg-gray-100 text-gray-400'
   293→                  }`}>
   294→                  <div className={`flex h-7 w-7 items-center justify-center rounded-full sm:h-8 sm:w-8 ${isCompleted ? 'bg-emerald-600 text-white' : ''}`}>
   295→                    {isCompleted ? <Check className="h-3.5 w-3.5 sm:h-4 sm:w-4" /> : <Icon className="h-3.5 w-3.5 sm:h-4 sm:w-4" />}
   296→                  </div>
   297→                  <span className="text-[9px] leading-tight font-medium text-center truncate w-full sm:text-[10px]">{step.title}</span>
   298→                  <span className="hidden text-[8px] leading-tight text-center opacity-75 truncate w-full sm:block sm:text-[9px]">{step.titleOr}</span>
   299→                </button>
   300→                {idx < STEPS.length - 1 && (
   301→                  <div className={`mx-0.5 h-0.5 flex-1 rounded sm:mx-1 ${currentStep > step.id ? 'bg-emerald-500' : 'bg-gray-200'}`} />
   302→                )}
   303→              </div>
   304→            );
   305→          })}
   306→        </div>
   307→        <div className="mt-1.5 flex items-center justify-between text-[11px] text-muted-foreground sm:mt-2 sm:text-xs">
   308→          <span>Step {currentStep} of {STEPS.length} / Qajeelfannaa {currentStep} kiyya {STEPS.length} keessaa</span>
   309→          <button type="button" onClick={resetForm} className="flex items-center gap-1 text-red-500 hover:text-red-700">
   310→            <RotateCcw className="h-3 w-3" />
   311→            <span className="hidden sm:inline">Reset / Balleessuu</span><span className="sm:hidden">Reset</span>
   312→          </button>
   313→        </div>
   314→      </div>
   315→
   316→      <form onSubmit={handleSubmit}>
   317→        <Card className="min-h-[300px] sm:min-h-[360px]">
   318→          {/* Step 1: Establishment */}
   319→          {currentStep === 1 && (
   320→            <>
   321→              <CardHeader className="pb-2 sm:pb-3">
   322→                <CardTitle className="flex items-center gap-1.5 text-base sm:gap-2 sm:text-lg">
   323→                  <Building2 className="h-4 w-4 shrink-0 text-emerald-600 sm:h-5 sm:w-5" />
   324→                  <span className="leading-tight">Odeeffannoo Hundeeffama<br /><span className="text-sm font-normal text-muted-foreground sm:text-base">Establishment information</span></span>
   325→                </CardTitle>
   326→                <CardDescription className="text-xs sm:text-sm">Basic details about the establishment</CardDescription>
   327→              </CardHeader>
   328→              <CardContent className="space-y-3 sm:space-y-4">
   329→                <div className="space-y-1.5 sm:space-y-2">
   330→                  <FieldLabel id="organizationName" en={label('organizationName').en} or={label('organizationName').or} required={req('organizationName')} />
   331→                  <Input id="organizationName" name="organizationName" placeholder="e.g., GOLD MARK HOTEL" value={formData.organizationName} onChange={handleChange} required className="text-sm sm:text-base" />
   332→                </div>
   333→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 sm:gap-4">
   334→                  <div className="space-y-1.5 sm:space-y-2">
   335→                    <FieldLabel id="subCity" en={label('subCity').en} or={label('subCity').or} required={req('subCity')} />
   336→                    <Select value={formData.subCity} onValueChange={handleSubCityChange}>
   337→                      <SelectTrigger id="subCity" className="text-sm sm:text-base"><SelectValue placeholder="Select / Filadhu" /></SelectTrigger>
   338→                      <SelectContent>{subCities.map((sc) => (<SelectItem key={sc.name} value={sc.name}>{sc.name}</SelectItem>))}</SelectContent>
   339→                    </Select>
   340→                  </div>
   341→                  <div className="space-y-1.5 sm:space-y-2">
   342→                    <FieldLabel id="area" en={label('area').en} or={label('area').or} required={req('area')} />
   343→                    <Select value={formData.area} onValueChange={(v) => handleSelectChange('area', v)} disabled={!selectedSubCity}>
   344→                      <SelectTrigger id="area" className="text-sm sm:text-base"><SelectValue placeholder={selectedSubCity ? 'Select / Filadhu' : 'Select sub-city first'} /></SelectTrigger>
   345→                      <SelectContent>{areas.map((a) => (<SelectItem key={a} value={a}>{a}</SelectItem>))}</SelectContent>
   346→                    </Select>
   347→                  </div>
   348→                </div>
   349→                {shown('specificAddress') && (
   350→                <div className="space-y-1.5 sm:space-y-2">
   351→                  <FieldLabel id="specificAddress" en={label('specificAddress').en} or={label('specificAddress').or} required={req('specificAddress')} />
   352→                  <Input id="specificAddress" name="specificAddress" placeholder="Street name, landmark..." value={formData.specificAddress} onChange={handleChange} className="text-sm sm:text-base" />
   353→                </div>
   354→                )}
   355→              </CardContent>
   356→            </>
   357→          )}
   358→
   359→          {/* Step 2: License */}
   360→          {currentStep === 2 && (
   361→            <>
   362→              <CardHeader className="pb-2 sm:pb-3">
   363→                <CardTitle className="flex items-center gap-1.5 text-base sm:gap-2 sm:text-lg">
   364→                  <FileText className="h-4 w-4 shrink-0 text-blue-600 sm:h-5 sm:w-5" />
   365→                  <span className="leading-tight">Dandeettii fi Hayyama<br /><span className="text-sm font-normal text-muted-foreground sm:text-base">Capacity and license</span></span>
   366→                </CardTitle>
   367→                <CardDescription className="text-xs sm:text-sm">Room capacity and licensing information</CardDescription>
   368→              </CardHeader>
   369→              <CardContent className="space-y-3 sm:space-y-4">
   370→                <div className="space-y-1.5 sm:space-y-2">
   371→                  <FieldLabel id="numberOfRooms" en={label('numberOfRooms').en} or={label('numberOfRooms').or} required={req('numberOfRooms')} />
   372→                  <Input id="numberOfRooms" name="numberOfRooms" type="number" min="1" placeholder="e.g., 20" value={formData.numberOfRooms} onChange={handleChange} required className="text-sm sm:text-base" />
   373→                </div>
   374→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 sm:gap-4">
   375→                  {shown('licenseType') && (
   376→                  <div className="space-y-1.5 sm:space-y-2">
   377→                    <FieldLabel id="licenseType" en={label('licenseType').en} or={label('licenseType').or} required={req('licenseType')} />
   378→                    <Select value={formData.licenseType} onValueChange={(v) => handleSelectChange('licenseType', v)}>
   379→                      <SelectTrigger className="text-sm sm:text-base"><SelectValue placeholder="Select / Filadhu" /></SelectTrigger>
   380→                      <SelectContent>{LICENSE_TYPES.map((t) => (<SelectItem key={t} value={t}>{t}</SelectItem>))}</SelectContent>
   381→                    </Select>
   382→                  </div>
   383→                  )}
   384→                  {shown('licenseLevel') && (
   385→                  <div className="space-y-1.5 sm:space-y-2">
   386→                    <FieldLabel id="licenseLevel" en={label('licenseLevel').en} or={label('licenseLevel').or} required={req('licenseLevel')} />
   387→                    <Select value={formData.licenseLevel} onValueChange={(v) => handleSelectChange('licenseLevel', v)}>
   388→                      <SelectTrigger className="text-sm sm:text-base"><SelectValue placeholder="Select / Filadhu" /></SelectTrigger>
   389→                      <SelectContent>{LICENSE_LEVELS.map((l) => (<SelectItem key={l} value={l}>{l}</SelectItem>))}</SelectContent>
   390→                    </Select>
   391→                  </div>
   392→                  )}
   393→                </div>
   394→                {shown('licenseNumber') && (
   395→                <div className="space-y-1.5 sm:space-y-2">
   396→                  <FieldLabel id="licenseNumber" en={label('licenseNumber').en} or={label('licenseNumber').or} required={req('licenseNumber')} />
   397→                  <Input id="licenseNumber" name="licenseNumber" placeholder="e.g., GH-2024-001" value={formData.licenseNumber} onChange={handleChange} className="text-sm sm:text-base" />
   398→                </div>
   399→                )}
   400→              </CardContent>
   401→            </>
   402→          )}
   403→
   404→          {/* Step 3: Contact */}
   405→          {currentStep === 3 && (
   406→            <>
   407→              <CardHeader className="pb-2 sm:pb-3">
   408→                <CardTitle className="flex items-center gap-1.5 text-base sm:gap-2 sm:text-lg">
   409→                  <Phone className="h-4 w-4 shrink-0 text-purple-600 sm:h-5 sm:w-5" />
   410→                  <span className="leading-tight">Contact Information<br /><span className="text-sm font-normal text-muted-foreground sm:text-base">Odeeffannoo Quunnamtii</span></span>
   411→                </CardTitle>
   412→                <CardDescription className="text-xs sm:text-sm">Owner and contact details</CardDescription>
   413→              </CardHeader>
   414→              <CardContent className="space-y-3 sm:space-y-4">
   415→                {shown('ownerName') && (
   416→                <div className="space-y-1.5 sm:space-y-2">
   417→                  <FieldLabel id="ownerName" en={label('ownerName').en} or={label('ownerName').or} required={req('ownerName')} />
   418→                  <Input id="ownerName" name="ownerName" placeholder="Full name of owner" value={formData.ownerName} onChange={handleChange} className="text-sm sm:text-base" />
